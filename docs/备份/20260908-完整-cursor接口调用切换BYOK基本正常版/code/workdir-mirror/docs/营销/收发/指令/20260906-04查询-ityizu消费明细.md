# 04 查询 ityizu@foxmail.com 消费明细，按模型拆分
# 生产库连接串从 .env 读取

$ErrorActionPreference = "Stop"
cd C:\ai24x01

# 从 .env 取 DATABASE_URL
$envContent = Get-Content api\.env -Raw
$match = [regex]::Match($envContent, 'DATABASE_URL=(\S+)')
if (-not $match.Success) { throw "DATABASE_URL not found in .env" }
$dbUrl = $match.Groups[1].Value

# 先查 auth_user_id
$userIdQuery = "SELECT id, email FROM auth_user WHERE email = 'ityizu@foxmail.com'"
$userId = & "C:\AI24X\postgresql\pgsql\bin\psql" -d $dbUrl -t -A -c $userIdQuery 2>&1
Write-Host "User lookup: $userId"

# 按模型分组汇总
$query = @"
SELECT 
    model,
    COUNT(*) AS calls,
    ROUND(SUM(prompt_tokens)::numeric / 1000000, 2) AS input_M,
    ROUND(SUM(completion_tokens)::numeric / 1000000, 2) AS output_M,
    ROUND(SUM(amount_usd)::numeric / 100, 2) AS total_usd
FROM billing_ledger
WHERE auth_user_id = (SELECT id FROM auth_user WHERE email = 'ityizu@foxmail.com')
  AND created_at >= NOW() - INTERVAL '7 days'
GROUP BY model
ORDER BY SUM(amount_usd) DESC;
"@

$result = & "C:\AI24X\postgresql\pgsql\bin\psql" -d $dbUrl -t -A -c $query 2>&1
Write-Host "`n=== 按模型消费明细 ==="
$result

# 总计
$totalQuery = @"
SELECT 
    ROUND(SUM(amount_usd)::numeric / 100, 2) AS total_usd,
    ROUND(SUM(prompt_tokens)::numeric / 1000000, 2) AS total_input_M,
    ROUND(SUM(completion_tokens)::numeric / 1000000, 2) AS total_output_M
FROM billing_ledger
WHERE auth_user_id = (SELECT id FROM auth_user WHERE email = 'ityizu@foxmail.com')
  AND created_at >= NOW() - INTERVAL '7 days';
"@

$total = & "C:\AI24X\postgresql\pgsql\bin\psql" -d $dbUrl -t -A -c $totalQuery 2>&1
Write-Host "`n=== 总计 ==="
$total
