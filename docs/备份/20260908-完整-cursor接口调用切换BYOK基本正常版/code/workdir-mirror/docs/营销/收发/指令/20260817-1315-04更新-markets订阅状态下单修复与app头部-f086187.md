function Invoke-Code($url, $method, $body, $token) {
    try {
        $hdrs = @{}
        if ($token) { $hdrs.Authorization = "Bearer $token" }
        $args = @{ Uri = $url; Method = $method; TimeoutSec = 20; UseBasicParsing = $true }
        if ($body) { $args.ContentType = "application/json"; $args.Body = $body }
        if ($hdrs.Count -gt 0) { $args.Headers = $hdrs }
        $r = Invoke-WebRequest @args
        return ("{0}" -f [int]$r.StatusCode)
    } catch {
        return ("{0}" -f [int]$_.Exception.Response.StatusCode)
    }
}
