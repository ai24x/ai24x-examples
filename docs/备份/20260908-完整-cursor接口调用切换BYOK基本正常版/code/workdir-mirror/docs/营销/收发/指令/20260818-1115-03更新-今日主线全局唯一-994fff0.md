## 执行步骤（PowerShell，整段复制运行）
```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "994fff0"

Set-Location $REPO

"--- 0) 备份本地脏改动再还原（防止 pull 冲突）---"
$FILES = @("p\a1\web\js\bjscreener.js","p\a1\web\gd.html","p\a1\web\m\gd.html")
$BK = "C:\backup\ai24x_a\20260818_pre_994fff0"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  if (Test-Path "$REPO\$x") {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) { Copy-Item "$REPO\$x" "$BK\$(Split-Path $x -Leaf).bak" -Force; git -C $REPO checkout -- $x }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记检查（注意：PowerShell 单引号内勿写 \`" 转义，直接写双引号字符） ---"
$js = Get-Content "$REPO\p\a1\web\js\bjscreener.js" -Raw -Encoding UTF8
$gd = Get-Content "$REPO\p\a1\web\gd.html" -Raw -Encoding UTF8
$m  = Get-Content "$REPO\p\a1\web\m\gd.html" -Raw -Encoding UTF8
"js_isTop=$($js.Contains('var isTop = !!'))  js_topMl=$($js.Contains('topMlName'))  js_market=$($js.Contains('市场强势'))"
"gd_v82=$($gd.Contains('bjscreener.js?v=82'))"
"m_isTop=$($m.Contains('var isTop = !!'))  m_topMl=$($m.Contains('topMlName'))"
if (-not ($js.Contains('var isTop = !!') -and $js.Contains('topMlName') -and $gd.Contains('bjscreener.js?v=82') -and $m.Contains('var isTop = !!'))) {
  Write-Host "!! 文件标记缺失" -ForegroundColor Red; exit 1
}

"--- 3) 重启 AI24X-a1-api (8001) ---"
$before = (Get-Process -Name python -ErrorAction SilentlyContinue | Sort-Object StartTime -Descending | Select-Object -First 1).StartTime
Restart-Service $SRV -Force
$ok = $false
for ($i = 0; $i -lt 15; $i++) {
  Start-Sleep -Seconds 5
  try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing; $ok = $true; break } catch {}
}
if (-not $ok) { Write-Host "!! /health 未就绪" -ForegroundColor Red; exit 1 }
"health=$($h.status)"
$after = (Get-Process -Name python -ErrorAction SilentlyContinue | Sort-Object StartTime -Descending | Select-Object -First 1).StartTime
"proc_before=$before"
"proc_after =$after"
if ($after -le $before) { Write-Host "!! 进程新建时间未更新（Restart-Service 可能未真正重启，需手动补重启）" -ForegroundColor Red; exit 1 }

"--- 4) 公网页面内容标记 ---"
$pubGd = (Invoke-WebRequest -Uri "https://a.ai24x.com/gd.html" -UseBasicParsing -TimeoutSec 20).Content
$pubJs = (Invoke-WebRequest -Uri "https://a.ai24x.com/js/bjscreener.js?v=82" -UseBasicParsing -TimeoutSec 20).Content
$pubM  = (Invoke-WebRequest -Uri "https://a.ai24x.com/m/gd.html" -UseBasicParsing -TimeoutSec 20).Content
"pub_gd_v82=$($pubGd.Contains('bjscreener.js?v=82'))"
"pub_js_isTop=$($pubJs.Contains('var isTop = !!'))"
"pub_m_isTop=$($pubM.Contains('var isTop = !!'))"
if (-not ($pubGd.Contains('bjscreener.js?v=82') -and $pubJs.Contains('var isTop = !!') -and $pubM.Contains('var isTop = !!'))) {
  Write-Host "!! 公网内容不对" -ForegroundColor Red; exit 1
}

Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
```
