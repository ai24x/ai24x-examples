# 【03 更新】推荐位硬顶 2 只（精选满则不展示观察）

> 通道：司令直连 03（deploy03.ps1）｜ 目标提交 **EXP=`528e14120b30`**
> 03 主机：123.207.199.238 · 仓库 `C:\ai24x01` · 服务 `AI24X-a1-api`（NSSM，8001）

## 背景
上一版精选上限 2，但仍可能再渲染「观察」池，页面看起来像推了 3 只。改为推荐位合计硬顶 2：精选优先，观察仅补空位；精选已满则不返回/不展示观察。静态 `bjscreener.js?v=151`。

## 执行步骤
```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "528e14120b30"
Set-Location $REPO

$FILES = @(
  "p\a1\web\gd.html",
  "p\a1\web\js\bjscreener.js",
  "p\a1\api\server\app\bj_screener.py"
)
$BK = "C:\backup\ai24x_a\$(Get-Date -Format yyyyMMdd)_pre_pick2_only_$EXP"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  $full = Join-Path $REPO $x
  if (Test-Path $full) {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) {
      Copy-Item $full (Join-Path $BK ((Split-Path $x -Leaf) + ".bak")) -Force
      git -C $REPO checkout -- $x
    }
  }
}

git pull origin master
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含 $EXP" -ForegroundColor Red; exit 1 }

$gd = Get-Content (Join-Path $REPO "p\a1\web\gd.html") -Raw -Encoding UTF8
if (-not $gd.Contains("bjscreener.js?v=151")) { Write-Host "!! 缺 v=151" -ForegroundColor Red; exit 1 }
$js = Get-Content (Join-Path $REPO "p\a1\web\js\bjscreener.js") -Raw -Encoding UTF8
if ($js -notmatch 'watchSlots\s*=\s*Math\.max\(0,\s*2\s*-\s*picks\.length\)') { Write-Host "!! 缺 watchSlots 硬顶" -ForegroundColor Red; exit 1 }
$bj = Get-Content (Join-Path $REPO "p\a1\api\server\app\bj_screener.py") -Raw -Encoding UTF8
if (-not $bj.Contains("_watch_cap = max(0, 2 - len(picks))")) { Write-Host "!! 缺后端 _watch_cap" -ForegroundColor Red; exit 1 }

$beforePid = (Get-CimInstance Win32_Service -Filter "Name='$SRV'").ProcessId
Restart-Service $SRV -Force
$ok = $false; $afterPid = $beforePid
for ($i = 0; $i -lt 20; $i++) {
  Start-Sleep -Seconds 5
  try {
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
    if ($svc) { $afterPid = $svc.ProcessId }
    $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
    if ($afterPid -gt 0 -and $afterPid -ne $beforePid -and $h.ok) { $ok = $true; break }
  } catch {}
}
if (-not $ok) { Write-Host "!! 重启失败" -ForegroundColor Red; exit 1 }

$pub = Invoke-WebRequest -Uri "https://a.ai24x.com/gd.html" -UseBasicParsing -TimeoutSec 25
if ($pub.StatusCode -ne 200 -or $pub.Content -notmatch 'bjscreener\.js\?v=151') { Write-Host "!! 公网非 v=151" -ForegroundColor Red; exit 1 }
$jsPub = Invoke-WebRequest -Uri "https://a.ai24x.com/js/bjscreener.js?v=151" -UseBasicParsing -TimeoutSec 25
if ($jsPub.Content -notmatch 'watchSlots') { Write-Host "!! 公网 js 缺 watchSlots" -ForegroundColor Red; exit 1 }

Write-Host "OK HEAD=$HEAD EXP=$EXP v=151 watchSlots" -ForegroundColor Green
```
