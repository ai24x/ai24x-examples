$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "b09b7281e177"

    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) { Copy-Item "$REPO\$x" "$BK\$(Split-Path $x -Leaf).bak" -Force; git -C $REPO checkout -- $x }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"

$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记检查 ---"
$markers = @(
  @{ f = "p/a1/api/server/app/bj_screener.py"; m = "def _king_rank" },
  @{ f = "p/a1/api/server/app/bj_screener.py"; m = "firstWeek" },
  @{ f = "p/a1/api/server/app/daily_report.py"; m = "def _recent_plate_top5" },
  @{ f = "p/a1/api/server/app/daily_report.py"; m = "东方财富" },
  @{ f = "p/a1/web/js/bjscreener.js";           m = "筛选关注" },

  @{ f = "p/a1/web/m/gd.html";                  m = "东方财富" }
)
$fail = 0
foreach ($mk in $markers) {
  $f = Get-Content "$REPO\$($mk.f)" -Raw -Encoding UTF8
  $hit = $f.Contains($mk.m)
  "$($mk.f) -> $hit"

  Copy-Item $rep "$BK\report.cache.bak" -Force
  Remove-Item $rep -Force
  "report 缓存已备份并删除"

Push-Location "$REPO\p\a1\api\server"
python -m app.daily_report

  Copy-Item $bjcache "$BK\bj_scan_daily.json.bak" -Force

  "bj_scan_daily.json 已备份并清空"

  "bj_scan_daily.json 不存在（跳过）"

  }
