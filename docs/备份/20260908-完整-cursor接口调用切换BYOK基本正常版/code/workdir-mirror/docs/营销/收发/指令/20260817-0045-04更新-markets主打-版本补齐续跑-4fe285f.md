```powershell
# === AI24X 更新部署验收 · 目标 4fe285f（版本补齐续跑）===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV_WWW = "AI24X-core"
$EXP  = "4fe285f"

Set-Location $REPO

"--- 1) web 目录如有本地改动：备份后 checkout ---"
$dirtyWeb = git status --short -- web
if ($dirtyWeb) {
    Copy-Item "$REPO\web" "C:\Users\Administrator\ops\web-dirty-20260817-0045" -Recurse -Force
    git checkout -- web
    if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败" -ForegroundColor Red; exit 1 }
    "dirty backup + checkout: web"
} else {
    "clean: web"
}

"--- 2) 拉取并校验目标提交 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
if ($HEAD -notlike "$EXP*") { Write-Host "!! HEAD 不匹配，期望前缀 $EXP 实际 $HEAD" -ForegroundColor Red; exit 1 }
git diff --quiet $EXP HEAD -- web
if ($LASTEXITCODE -ne 0) { Write-Host "!! web 在 $EXP 之后仍有改动，停止" -ForegroundColor Red; exit 1 }
"target_verified=ok"

"--- 3) 重启 core (8002) ---"
Restart-Service $SRV_WWW -Force
$okW = $false
for ($i = 0; $i -lt 15; $i++) {
    Start-Sleep -Seconds 5
    try { $hw = Invoke-RestMethod -Uri "http://127.0.0.1:8002/health" -TimeoutSec 10 -UseBasicParsing; $okW = $true; break } catch {}
}
if (-not $okW) { Write-Host "!! core /health 未就绪" -ForegroundColor Red; exit 1 }
"core-health=$($hw.status) commit=$($hw.commit)"

"--- 4) 本地复验：旧版本号残留 0 + 关键页面内容 ---"
$oldVer = @(Get-ChildItem "$REPO\web" -Recurse -Filter *.html | Select-String -Pattern 'locales.js?v=20260818b' -SimpleMatch -List)
"local_old_ver_residual=$($oldVer.Count)"
if ($oldVer.Count -ne 0) { Write-Host "!! 旧版本号 20260818b 残留 $($oldVer.Count) 处" -ForegroundColor Red; exit 1 }
$oldShell = @(Get-ChildItem "$REPO\web" -Recurse -Filter *.html | Select-String -Pattern 'shell.js?v=20260818b' -SimpleMatch -List)
"local_old_shell_residual=$($oldShell.Count)"
if ($oldShell.Count -ne 0) { Write-Host "!! 旧 shell 版本号残留" -ForegroundColor Red; exit 1 }
$newVer = @(Get-ChildItem "$REPO\web" -Recurse -Filter *.html | Select-String -Pattern 'locales.js?v=20260818c' -SimpleMatch -List)
"local_new_ver_pages=$($newVer.Count)"
if ($newVer.Count -lt 45) { Write-Host "!! 新版页面数量异常（<45）" -ForegroundColor Red; exit 1 }

$pp = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/pricing.html?x=l4fe285f" -TimeoutSec 20 -UseBasicParsing).Content
"local_pricing_cards=$($pp.Contains('$24.9') -and $pp.Contains('$199'))"
"local_pricing_deeplink=$($pp.Contains('app.html?plan=yearly#sub'))"
$pi = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/index.html?x=l4fe285f" -TimeoutSec 20 -UseBasicParsing).Content
"local_index_startfree=$($pi.Contains('hero.cta.free'))"

"--- 5) 公网复验 ---"
$pub = (Invoke-WebRequest -Uri "https://www.ai24x.com/pricing.html?x=w4fe285f" -TimeoutSec 25 -UseBasicParsing).Content
"pub_pricing_cards=$($pub.Contains('$24.9') -and $pub.Contains('$199'))"
"pub_pricing_v18c=$($pub.Contains('locales.js?v=20260818c'))"
$pubA = (Invoke-WebRequest -Uri "https://www.ai24x.com/404.html?x=w4fe285f" -TimeoutSec 25 -UseBasicParsing).Content
"pub_404_v18c=$($pubA.Contains('locales.js?v=20260818c'))"
$pubG = (Invoke-WebRequest -Uri "https://www.ai24x.com/guides/index.html?x=w4fe285f" -TimeoutSec 25 -UseBasicParsing).Content
"pub_guides_v18c=$($pubG.Contains('../config/locales.js?v=20260818c'))"
$pubM = (Invoke-WebRequest -Uri "https://markets.ai24x.com/app.html?x=w4fe285f" -TimeoutSec 25 -UseBasicParsing).Content
"pub_markets_flash=$($pubM.Contains('function flashPlan'))"

if ($oldVer.Count -eq 0 -and $oldShell.Count -eq 0 -and $newVer.Count -ge 45 -and $pp.Contains('$24.9') -and $pp.Contains('app.html?plan=yearly#sub') -and $pi.Contains('hero.cta.free') -and $pub.Contains('$24.9') -and $pub.Contains('locales.js?v=20260818c') -and $pubA.Contains('locales.js?v=20260818c') -and $pubG.Contains('../config/locales.js?v=20260818c') -and $pubM.Contains('function flashPlan')) {
    "--- 全部 PASS ---"
    Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
} else {
    Write-Host "!! 验收未全通过" -ForegroundColor Red; exit 1
}
```
