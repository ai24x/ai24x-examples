## 背景
- 14db525 部署验收时 04 反馈：`get_kline_rows` 缓存命中条件用 `len(rows) >= min(count,30)`，250 根时代的旧缓存会让 count=500 请求只返回 250 根（验收首轮失败，04 清缓存绕行）。
- 本批修复（32eec0d，仅 `p/markets/api/server/app/providers_us.py`）：
  1. 缓存拉取统一用 `max(count, 500)` 根；
  2. 命中条件改为 `len(rows) >= count` 或 `complete=True`（上游历史已取完，短历史标的避免反复重拉）；
  3. 旧 250 根缓存无 complete 标记 → count=500 请求自动 miss 重拉，自愈无需手动清缓存。
- 本地已测：250 根污染缓存 → count=500 重拉 500 根；count=250 → 切片 250；count=600 → 重拉 600；短历史 complete 缓存命中不重拉。

## 注意事项
- 只动 1 个路径：`p/markets/api/server/app/providers_us.py`；04 工作树其它历史未提交文件一律不动。
- 无需手动清缓存（修复自愈）；服务重启后首次请求按需重拉。

"--- 1) providers_us.py 如有本地改动：备份后 checkout ---"

    git checkout -- $FILE
    if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败" -ForegroundColor Red; exit 1 }

} else {
    "clean: $FILE"
}

"--- 2) 拉取并校验 HEAD ---"

if ($HEAD -notlike "$EXP*") { Write-Host "!! HEAD 不匹配，期望前缀 $EXP 实际 $HEAD" -ForegroundColor Red; exit 1 }

"--- 3) 重启 markets (18012) ---"

for ($i = 0; $i -lt 12; $i++) {
    Start-Sleep -Seconds 5
    try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:18012/health" -TimeoutSec 10 -UseBasicParsing; $ok = $true; break } catch {}
}
if (-not $ok) { Write-Host "!! markets /health 未就绪" -ForegroundColor Red; exit 1 }
"markets-health=$($h.status)"

"--- 4) 本地缓存修复验收 ---"

"local_count700=$($k700.data.candles.Count)"
if ($k700.data.candles.Count -ne 700) { Write-Host "!! count=700 未重拉（缓存未按 count 区分）" -ForegroundColor Red; exit 1 }
$k250 = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=AAPL&count=250" -TimeoutSec 30 -UseBasicParsing
"local_count250=$($k250.data.candles.Count)"
if ($k250.data.candles.Count -ne 250) { Write-Host "!! count=250 切片异常" -ForegroundColor Red; exit 1 }
$w = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=WETOUR" -TimeoutSec 30 -UseBasicParsing
"local_wetour_suggested=$($w.suggested)"
if ($w.suggested -ne "WETO") { Write-Host "!! WETOUR 建议代码未命中" -ForegroundColor Red; exit 1 }

"--- 5) 公网验收 ---"
$pk = Invoke-RestMethod -Uri "https://markets.ai24x.com/api/kline?symbol=AAPL&count=500" -TimeoutSec 30 -UseBasicParsing
"pub_count500=$($pk.data.candles.Count)"
$pk2 = Invoke-RestMethod -Uri "https://markets.ai24x.com/api/kline?symbol=AAPL&count=250" -TimeoutSec 30 -UseBasicParsing
"pub_count250=$($pk2.data.candles.Count)"

if ($pk.data.candles.Count -eq 500 -and $pk2.data.candles.Count -eq 250) {
    "--- 全部 PASS ---"
    Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
} else {
    Write-Host "!! 验收未全通过" -ForegroundColor Red; exit 1
}
