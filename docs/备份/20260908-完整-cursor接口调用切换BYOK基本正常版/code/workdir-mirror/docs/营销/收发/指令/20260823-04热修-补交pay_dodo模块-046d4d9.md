# 【04 热修】补交 api/pay_dodo.py（生产 token-admin 登录 500 根因）

> 通道：司令直连 04（deploy04.ps1）｜ 目标提交 **046d4d9**
> 现象：生产 token-admin 登录报「无法进入：Failed to fetch」；核心日志 `ModuleNotFoundError: No module named 'pay_dodo'`（上一提交 26841d4 只带了 pay_dodo_products.py，漏了 pay_dodo.py）

## 执行步骤（PowerShell，整段复制运行）
```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-core"
$EXP  = "046d4d9"

Set-Location $REPO
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

if (-not (Test-Path "$REPO\api\pay_dodo.py")) { Write-Host "!! pay_dodo.py 不存在" -ForegroundColor Red; exit 1 }
$m = (Get-Content "$REPO\api\pay_dodo.py" -Raw -Encoding UTF8).Contains('def dodo_configured')
"mark_pay_dodo=$m"
if (-not $m) { Write-Host "!! pay_dodo.py 标记缺失" -ForegroundColor Red; exit 1 }

$beforePid = (Get-CimInstance Win32_Service -Filter "Name='$SRV'").ProcessId
"svc_pid_before=$beforePid"
Restart-Service $SRV -Force
$ok = $false
for ($i = 0; $i -lt 24; $i++) {
  Start-Sleep -Seconds 5
  try {
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
    $afterPid = $svc.ProcessId
    $h = Invoke-RestMethod -Uri "http://127.0.0.1:8002/health" -TimeoutSec 10 -UseBasicParsing
    if ($afterPid -gt 0 -and $afterPid -ne $beforePid -and $h.commit -and $h.commit -like "$EXP*") { $ok = $true; break }
  } catch {}
}
"svc_pid_after=$afterPid health_ok=$ok commit=$($h.commit)"
if (-not $ok) { Write-Host "!! 服务未按预期重启或 commit 不符" -ForegroundColor Red; exit 1 }

# 用 .env 真实 ADMIN_API_KEY 验证 summary 200（不打印密钥）
$key = (Select-String -Path "$REPO\api\.env" -Pattern "^ADMIN_API_KEY=(.+)$").Matches[0].Groups[1].Value.Trim()
$r = Invoke-WebRequest -Uri "http://127.0.0.1:8002/v1/admin/token/summary" -UseBasicParsing -TimeoutSec 25 -Headers @{ "X-Admin-Key" = $key } -ErrorAction Stop
"summary_status=" + $r.StatusCode
if ($r.StatusCode -ne 200) { Write-Host "!! summary 非 200" -ForegroundColor Red; exit 1 }

Write-Host "=== 热修部署成功 OK ===" -ForegroundColor Green
```
