## 副脑04 执行脚本（整段复制运行）
```powershell
# === AI24X 更新部署验收 · 目标 9d31a6e ===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-core"
$EXP  = "9d31a6e"

Set-Location $REPO

"--- 1) blog/index.html 本地改动备份+清理 ---"
$dirty = git status --short -- web/blog/index.html
if ($dirty) {
    Copy-Item "$REPO\web\blog\index.html" "C:\Users\Administrator\ops\blog-index-dirty-20260817-0005.html" -Force
    git checkout -- web/blog/index.html
    if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败" -ForegroundColor Red; exit 1 }
    "dirty backup + checkout: web/blog/index.html"
} else {
    "clean: web/blog/index.html"
}

"--- 2) 拉取并校验目标提交 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! 目标 $EXP 不在 HEAD 历史中" -ForegroundColor Red; exit 1 }
git diff --quiet $EXP HEAD -- web/blog/index.html
if ($LASTEXITCODE -ne 0) { Write-Host "!! web/blog/index.html 在 $EXP 之后仍有改动" -ForegroundColor Red; exit 1 }
"target_verified=ok"

"--- 3) 重启 core (8002) ---"
Restart-Service $SRV -Force
$cok = $false
for ($i = 0; $i -lt 12; $i++) {
    Start-Sleep -Seconds 5
    try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:8002/health" -TimeoutSec 10 -UseBasicParsing; $cok = $true; break } catch {}
}
if (-not $cok) { Write-Host "!! core /health 未就绪" -ForegroundColor Red; exit 1 }
"core-health=$($h.status) commit=$($h.commit)"

"--- 4) 本地验收 ---"
$blueResid = @(Get-ChildItem "$REPO\web" -Recurse -Filter *.html | Select-String -Pattern 'theme-blue' -List)
"local_theme_blue_residual=$($blueResid.Count)"
if ($blueResid.Count -ne 0) { Write-Host "!! web 仍有 theme-blue 残留" -ForegroundColor Red; exit 1 }
$darkPages = @(Get-ChildItem "$REPO\web" -Recurse -Filter *.html | Select-String -Pattern 'theme-dark.css' -List)
"local_dark_pages=$($darkPages.Count)"
if ($darkPages.Count -lt 46) { Write-Host "!! 深色页面数量异常（<46）" -ForegroundColor Red; exit 1 }
$bl = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/blog/index.html?x=l9d31a6e" -TimeoutSec 20 -UseBasicParsing).Content
"local_blog_dark=$($bl.Contains('theme-dark.css') -and $bl -match 'class=\"theme-dark\"')"
"local_blog_v18b=$($bl.Contains('locales.js?v=20260818b') -and $bl.Contains('shell.js?v=20260818b'))"
if (-not ($bl.Contains('theme-dark.css') -and $bl -match 'class=\"theme-dark\"')) { Write-Host "!! blog 未深色" -ForegroundColor Red; exit 1 }

"--- 5) 公网验收 ---"
$pub = (Invoke-WebRequest -Uri "https://www.ai24x.com/blog/index.html?x=w9d31a6e" -TimeoutSec 25 -UseBasicParsing).Content
"pub_blog_dark=$($pub.Contains('theme-dark.css') -and $pub -match 'class=\"theme-dark\"')"
"pub_blog_no_blue=$(-not $pub.Contains('theme-blue'))"
"pub_blog_v18b=$($pub.Contains('locales.js?v=20260818b'))"
$pubC = (Invoke-WebRequest -Uri "https://www.ai24x.com/pricing.html?x=w9d31a6e" -TimeoutSec 25 -UseBasicParsing).Content
"pub_pricing_no_blue=$(-not $pubC.Contains('theme-blue'))"

if ($blueResid.Count -eq 0 -and $darkPages.Count -ge 46 -and $bl.Contains('theme-dark.css') -and $pub.Contains('theme-dark.css') -and (-not $pub.Contains('theme-blue')) -and $pub.Contains('locales.js?v=20260818b') -and (-not $pubC.Contains('theme-blue'))) {
    "--- 全部 PASS ---"
    Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
} else {
    Write-Host "!! 验收未全通过" -ForegroundColor Red; exit 1
}
```
