## 副脑04 执行脚本（整段复制运行）
```powershell
# === AI24X 更新部署验收 · open 站文案清理与版本升级 · 目标 11c97a3 ===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$EXP  = "11c97a3"
$OPEN_SITE = "C:\nginx\html\ai24x"
$OPEN_SRC  = "C:\ai24x01\p\open\web"
$OPEN_BAK  = "C:\nginx\html\ai24x.bak-20260817-2005-文案清理"
$FILES = @(
  "p/open/web/config/locales.js",
  "p/open/web/pricing.html",
  "p/open/web/product.html",
  "p/markets/scripts/_qa_open_pages_20260817.js"
)

Set-Location $REPO

"--- 0) 拉取前工作树状态 ---"
git status --short | Select-Object -First 20

"--- 1) 本次路径如有本地改动：备份后 checkout ---"
foreach ($f in $FILES) {
    $dirty = git status --short -- $f
    if ($dirty) {
        $bn = ($f -replace "/", "_")
        Copy-Item "$REPO\$f" "C:\Users\Administrator\ops\dirty-$bn-20260817-2005" -Force
        git checkout -- $f
        if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败: $f" -ForegroundColor Red; exit 1 }
        "dirty backup + checkout: $f"
    } else {
        "clean: $f"
    }
}

"--- 2) 拉取并校验目标提交已包含、本次路径与目标一致 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! 目标 $EXP 不在 HEAD 历史中" -ForegroundColor Red; exit 1 }
git diff --quiet $EXP HEAD -- p/open/web p/markets/scripts/_qa_open_pages_20260817.js
if ($LASTEXITCODE -ne 0) { Write-Host "!! p/open 在 $EXP 之后仍有改动，停止" -ForegroundColor Red; exit 1 }
"target_verified=ok"

"--- 3) 备份现有 open 站目录 + robocopy 镜像同步 ---"
if (-not (Test-Path $OPEN_BAK)) {
    Copy-Item -LiteralPath $OPEN_SITE -Destination $OPEN_BAK -Recurse -Force
}
"backup=$OPEN_BAK"
if (-not (Test-Path "$OPEN_SRC\index.html")) { Write-Host "!! p\open\web 不存在 index.html" -ForegroundColor Red; exit 1 }
robocopy $OPEN_SRC $OPEN_SITE /MIR /NFL /NDL /NJH /NJS /NP
if ($LASTEXITCODE -ge 8) { Write-Host "!! robocopy 失败" -ForegroundColor Red; exit 1 }
"sync=OK"

"--- 4) 本地验收（站点目录）---"
$oldVer = (Get-ChildItem $OPEN_SITE -Recurse -Filter *.html | Where-Object { (Get-Content $_.FullName -Raw) -match 'v=20260816y' } | Measure-Object).Count
"old_ver_files=$oldVer"
if ($oldVer -ne 0) { Write-Host "!! 仍有旧版本号 20260816y 残留" -ForegroundColor Red; exit 1 }

$loc = Get-Content "$OPEN_SITE\config\locales.js" -Raw -Encoding UTF8
"loc_titleTag=$($loc.Contains('page.pricing.titleTag'))"
"loc_matrix_zh=$($loc.Contains('想调用中国名模：用 AI24X 统一 API'))"
"loc_matrix_en=$($loc.Contains(""Call China's leading models with one AI24X key""))"
"loc_callout_no_old=$((-not $loc.Contains('一次充值，用完为止')) -and (-not $loc.Contains('Pay once, use until empty')))"
if (-not ($loc.Contains('page.pricing.titleTag') -and $loc.Contains('想调用中国名模：用 AI24X 统一 API') -and $loc.Contains(""Call China's leading models with one AI24X key"") -and (-not $loc.Contains('一次充值，用完为止')) -and (-not $loc.Contains('Pay once, use until empty')))) {
    Write-Host "!! locales 文案校验未全过" -ForegroundColor Red; exit 1
}

$pr = Get-Content "$OPEN_SITE\pricing.html" -Raw -Encoding UTF8
"pricing_titleTag=$($pr.Contains('data-i18n="page.pricing.titleTag"'))"
"pricing_no_tip_zh=$((-not $pr.Contains('要点名选 Scale（资格+预充一次齐）')))"
"pricing_no_tip_en=$((-not $pr.Contains('Name models with Scale')))"
"pricing_v=$($pr.Contains('v=20260817z'))"
if (-not ($pr.Contains('data-i18n="page.pricing.titleTag"') -and (-not $pr.Contains('要点名选 Scale（资格+预充一次齐）')) -and (-not $pr.Contains('Name models with Scale')) -and $pr.Contains('v=20260817z'))) {
    Write-Host "!! pricing 校验未全过" -ForegroundColor Red; exit 1
}

"--- 5) 公网 open 站验收 ---"
foreach ($u in @('/product.html', '/pricing.html', '/models/vip-picks.html', '/console.html')) {
    try {
        $r = Invoke-WebRequest -Uri ("https://open.ai24x.com" + $u + "?x=open2005") -TimeoutSec 20 -UseBasicParsing
        "$u => $($r.StatusCode) v=$($r.Content.Contains('v=20260817z'))"
    } catch {
        Write-Host "!! $u 失败: $($_.Exception.Message)" -ForegroundColor Red; exit 1
    }
}
$ppub = (Invoke-WebRequest -Uri "https://open.ai24x.com/pricing.html?x=open2005" -TimeoutSec 20 -UseBasicParsing).Content
"pub_pricing_no_tip=$((-not $ppub.Contains('要点名选 Scale（资格+预充一次齐）')) -and (-not $ppub.Contains('Name models with Scale')))"
$vpub = (Invoke-WebRequest -Uri "https://open.ai24x.com/models/vip-picks.html?x=open2005" -TimeoutSec 20 -UseBasicParsing).Content
"pub_vippicks_ok=$($vpub.Contains('v=20260817z'))"

if ($ppub.Contains('要点名选 Scale（资格+预充一次齐）') -or $ppub.Contains('Name models with Scale') -or (-not $vpub.Contains('v=20260817z'))) {
    Write-Host "!! 公网验收未全过" -ForegroundColor Red; exit 1
}

"--- 全部 PASS ---"
Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
```
