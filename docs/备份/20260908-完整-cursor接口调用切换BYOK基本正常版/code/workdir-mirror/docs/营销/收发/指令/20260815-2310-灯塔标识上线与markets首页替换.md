```powershell
  @{n="www hero 灯塔徽标";      f={ $www.Contains("lighthouse-mark") }},
  @{n="www 版本 20260817a";     f={ $www.Contains("locales.js?v=20260817a") }},

  @{n="markets CTA 进行情工具"; f={ $mh.Contains('a class="btn btn-primary" href="/app.html"') }}
)
$fail = 0
foreach ($c in $checks) {
  $ok = & $c.f
  Write-Host ("{0}  {1}" -f ($(if ($ok) { "OK " } else { "FAIL" })), $c.n)
  if (-not $ok) { $fail++ }
}
if ($fail -gt 0) { Write-Host "!! 静态文件校验失败" -ForegroundColor Red; exit 1 }
```
