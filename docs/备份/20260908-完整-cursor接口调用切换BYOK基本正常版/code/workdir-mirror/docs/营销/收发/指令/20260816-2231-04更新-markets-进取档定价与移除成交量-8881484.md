# === AI24X 更新部署验收 · 目标 8881484 ===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV_MARKETS = "AI24X-markets-api"
$EXP  = "8881484"
$SITES = "C:\sites\markets.ai24x.com"
$FILES = @(
  "p/markets/api/server/app/billing.py",
  "p/markets/web/app.html",
  "p/markets/web/pricing.html"
)

Set-Location $REPO

"--- 1) 三个路径如有本地改动：备份后 checkout ---"
foreach ($f in $FILES) {
    $dirty = git status --short -- $f
    if ($dirty) {
        $bn = ($f -replace "/", "_")
        Copy-Item "$REPO\$f" "C:\Users\Administrator\ops\dirty-$bn-20260816-2231" -Force
        git checkout -- $f
        if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败: $f" -ForegroundColor Red; exit 1 }
        "dirty backup + checkout: $f"
    } else {
        "clean: $f"
    }
}

"--- 2) 拉取并校验目标提交已包含且 3 个路径与目标一致 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! 目标 $EXP 不在 HEAD 历史中" -ForegroundColor Red; exit 1 }
git diff --quiet $EXP HEAD -- p/markets/api/server/app/billing.py p/markets/web/app.html p/markets/web/pricing.html
if ($LASTEXITCODE -ne 0) { Write-Host "!! 3 个路径在 $EXP 之后仍有改动，停止" -ForegroundColor Red; exit 1 }
"target_verified=ok"

"--- 3) 同步静态文件到 nginx 服务目录（先备份）---"
foreach ($f in @("app.html", "pricing.html")) {
    if (Test-Path "$SITES\$f") { Copy-Item "$SITES\$f" "$SITES\$f.bak-20260816-2231" -Force }
    Copy-Item "$REPO\p\markets\web\$f" "$SITES\$f" -Force
}

"--- 4) 重启 markets (18012) ---"
Restart-Service $SRV_MARKETS -Force
$ok = $false
for ($i = 0; $i -lt 12; $i++) {
    Start-Sleep -Seconds 5
    try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:18012/health" -TimeoutSec 10 -UseBasicParsing; $ok = $true; break } catch {}
}
if (-not $ok) { Write-Host "!! markets /health 未就绪" -ForegroundColor Red; exit 1 }
"markets-health=$($h.status)"

"--- 5) 本地验收 ---"
$b = python -c "import sys; sys.path.insert(0,r'p/markets/api/server'); from app.billing import PLANS; print(PLANS['monthly']['usd'], PLANS['yearly']['usd'])"
"local_billing=$b"
if ($b -ne "24.9 199.0") { Write-Host "!! billing 权威金额非 24.9/199" -ForegroundColor Red; exit 1 }

$sa = Get-Content "$SITES\app.html" -Raw -Encoding UTF8
"local_app_price24=$($sa.Contains('$24.9'))"
"local_app_price199=$($sa.Contains('$199'))"
"local_app_no_volcard=$((-not $sa.Contains('chart-vol')) -and (-not $sa.Contains('pan-vol')) -and (-not $sa.Contains('renderVol')))"
"local_app_lang_switch=$($sa.Contains('lang-switch'))"
$sp = Get-Content "$SITES\pricing.html" -Raw -Encoding UTF8
"local_pricing_new=$($sp.Contains('$24.9') -and $sp.Contains('$199'))"
"local_pricing_no_old=$((-not $sp.Contains('$14.9')) -and (-not $sp.Contains('$119')))"

$k = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=AAPL" -TimeoutSec 30 -UseBasicParsing
"local_kline_candles=$($k.data.candles.Count)"
if ($k.data.candles.Count -ne 500) { Write-Host "!! 本地 kline 非 500 根" -ForegroundColor Red; exit 1 }

"--- 6) 公网验收 ---"
$pub = (Invoke-WebRequest -Uri "https://markets.ai24x.com/app.html?x=agg8881484" -TimeoutSec 25 -UseBasicParsing).Content
"pub_app_price24=$($pub.Contains('$24.9'))"
"pub_app_price199=$($pub.Contains('$199'))"
"pub_app_no_volcard=$((-not $pub.Contains('chart-vol')) -and (-not $pub.Contains('pan-vol')))"
$ppub = (Invoke-WebRequest -Uri "https://markets.ai24x.com/pricing.html?x=agg8881484" -TimeoutSec 25 -UseBasicParsing).Content
"pub_pricing_new=$($ppub.Contains('$24.9') -and $ppub.Contains('$199'))"
"pub_pricing_no_old=$((-not $ppub.Contains('$14.9')) -and (-not $ppub.Contains('$119')))"
$pk = Invoke-RestMethod -Uri "https://markets.ai24x.com/api/kline?symbol=AAPL" -TimeoutSec 20 -UseBasicParsing
"pub_kline_candles=$($pk.data.candles.Count)"

if ($b -eq "24.9 199.0" -and $sa.Contains('$24.9') -and $sa.Contains('$199') -and (-not $sa.Contains('chart-vol')) -and (-not $sa.Contains('pan-vol')) -and $sp.Contains('$24.9') -and $sp.Contains('$199') -and (-not $sp.Contains('$14.9')) -and (-not $sp.Contains('$119')) -and $k.data.candles.Count -eq 500 -and $pub.Contains('$24.9') -and (-not $pub.Contains('chart-vol')) -and $ppub.Contains('$24.9') -and (-not $ppub.Contains('$14.9')) -and $pk.data.candles.Count -eq 500) {
    "--- 全部 PASS ---"
    Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
} else {
    Write-Host "!! 验收未全通过" -ForegroundColor Red; exit 1
}
