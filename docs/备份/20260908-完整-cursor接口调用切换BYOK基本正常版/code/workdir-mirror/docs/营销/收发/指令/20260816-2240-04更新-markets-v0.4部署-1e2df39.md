## 副脑04 执行脚本（整段复制运行）
```powershell
# === AI24X 更新部署验收 · 目标 1e2df39c5612 ===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV_MARKETS = "AI24X-markets-api"
$EXP  = "1e2df39c5612"
$SITES = "C:\sites\markets.ai24x.com"
$FILES = @(
  "p/markets/api/server/app/main.py",
  "p/markets/api/server/app/paypal.py",
  "p/markets/api/server/app/providers_us.py",
  "p/markets/web/app.html",
  "p/markets/web/index.html",
  "p/markets/web/pricing.html"
)

Set-Location $REPO

"--- 1) 六个路径如有本地改动：备份后 checkout ---"
foreach ($f in $FILES) {
    $dirty = git status --short -- $f
    if ($dirty) {
        $bn = ($f -replace "/", "_")
        Copy-Item "$REPO\$f" "C:\Users\Administrator\ops\dirty-$bn-20260816-2240" -Force
        git checkout -- $f
        if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败: $f" -ForegroundColor Red; exit 1 }
        "dirty backup + checkout: $f"
    } else {
        "clean: $f"
    }
}

"--- 2) 拉取并校验 HEAD ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
if ($HEAD -notlike "$EXP*") { Write-Host "!! HEAD 不匹配，期望前缀 $EXP 实际 $HEAD" -ForegroundColor Red; exit 1 }

"--- 3) 同步静态文件到 nginx 服务目录（先备份）---"
foreach ($f in @("app.html", "index.html", "pricing.html")) {
    if (Test-Path "$SITES\$f") { Copy-Item "$SITES\$f" "$SITES\$f.bak-20260816-2240" -Force }
    Copy-Item "$REPO\p\markets\web\$f" "$SITES\$f" -Force
}
$sa = Get-Content "$SITES\app.html" -Raw -Encoding UTF8
"sites_reset_tomorrow=$($sa.Contains('Free briefs reset tomorrow'))"
"sites_pro_activated=$($sa.Contains('Pro activated'))"
$si = Get-Content "$SITES\index.html" -Raw -Encoding UTF8
"sites_index_pricing_link=$($si.Contains('/pricing.html'))"
$sp = Get-Content "$SITES\pricing.html" -Raw -Encoding UTF8
"sites_pricing_price=$($sp.Contains('14.9') -and $sp.Contains('119'))"

"--- 4) 重启 markets (18012) ---"
Restart-Service $SRV_MARKETS -Force
$ok = $false
for ($i = 0; $i -lt 12; $i++) {
    Start-Sleep -Seconds 5
    try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:18012/health" -TimeoutSec 10 -UseBasicParsing; $ok = $true; break } catch {}
}
if (-not $ok) { Write-Host "!! markets /health 未就绪" -ForegroundColor Red; exit 1 }
"markets-health=$($h.status)"

"--- 5) PayPal env 核查（有覆盖变量即停）---"
$envPay = Select-String -Path "$REPO\api\.env", "$REPO\p\markets\api\server\.env" -Pattern "MARKETS_PAYPAL" -SimpleMatch -ErrorAction SilentlyContinue
"paypal_env_overrides=$($envPay.Count)"
if ($envPay.Count -gt 0) { Write-Host "!! 检测到 MARKETS_PAYPAL_* 覆盖变量，需删除后重启，停步" -ForegroundColor Red; exit 1 }

"--- 6) 本地 + 公网验收 ---"
$k = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=AAPL" -TimeoutSec 20 -UseBasicParsing
"local_kline_candles=$($k.data.candles.Count)"
if ($k.data.candles.Count -ne 250) { Write-Host "!! 本地 kline 非 250 根" -ForegroundColor Red; exit 1 }
$w = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=WETOUR" -TimeoutSec 30 -UseBasicParsing
"local_wetour_suggested=$($w.suggested)"
if ($w.suggested -ne "WETO") { Write-Host "!! WETOUR 建议代码未命中" -ForegroundColor Red; exit 1 }

$pub = (Invoke-WebRequest -Uri "https://markets.ai24x.com/app.html?x=v04" -TimeoutSec 25 -UseBasicParsing).Content
"pub_reset_tomorrow=$($pub.Contains('Free briefs reset tomorrow'))"
"pub_pro_activated=$($pub.Contains('Pro activated'))"
$ppub = (Invoke-WebRequest -Uri "https://markets.ai24x.com/pricing.html?x=v04" -TimeoutSec 25 -UseBasicParsing).Content
"pub_pricing_price=$($ppub.Contains('14.9') -and $ppub.Contains('119'))"
$pk = Invoke-RestMethod -Uri "https://markets.ai24x.com/api/kline?symbol=AAPL" -TimeoutSec 20 -UseBasicParsing
"pub_kline_candles=$($pk.data.candles.Count)"

if ($pub.Contains('Free briefs reset tomorrow') -and $pub.Contains('Pro activated') -and $ppub.Contains('14.9') -and $ppub.Contains('119') -and $pk.data.candles.Count -eq 250) {
    "--- 全部 PASS ---"
    Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
} else {
    Write-Host "!! 验收未全通过" -ForegroundColor Red; exit 1
}
```
