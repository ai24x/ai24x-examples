```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "5f47729"

Set-Location $REPO

"--- 0) 本地脏改动备份再还原（防止 pull 冲突）---"
$FILES = @("p\a1\web\m\gd.html")
$BK = "C:\backup\ai24x_a\20260818_pre_5f47729"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  if (Test-Path "$REPO\$x") {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) { Copy-Item "$REPO\$x" "$BK\$(Split-Path $x -Leaf).bak" -Force; git -C $REPO checkout -- $x }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验（目标提交在历史中）---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记检查（新内容在、旧内容不在）---"
$m = Get-Content "$REPO\p\a1\web\m\gd.html" -Raw -Encoding UTF8
"m_basis_fn=$($m.Contains('renderMainlineBasis'))  m_judgment=$($m.Contains('判定依据'))"
"m_star=$($m.Contains('评分最高'))  m_today=$($m.Contains('今日主线'))  m_break=$($m.Contains('破位占比'))"
"m_no_king=$(-not $m.Contains('王者'))  m_no_oldwin=$(-not $m.Contains('10日止损率'))  m_no_posspan=$(-not $m.Contains('class=\"pos\"'))"
if (-not ($m.Contains('renderMainlineBasis') -and $m.Contains('判定依据') -and $m.Contains('评分最高') -and $m.Contains('今日主线') -and $m.Contains('破位占比') -and (-not $m.Contains('王者')) -and (-not $m.Contains('10日止损率')))) {
  Write-Host "!! 文件标记缺失" -ForegroundColor Red; exit 1
}

"--- 3) 重启 AI24X-a1-api (8001) ---"
$before = (Get-Process -Name python -ErrorAction SilentlyContinue | Sort-Object StartTime -Descending | Select-Object -First 1).StartTime
Restart-Service $SRV -Force
$ok = $false
for ($i = 0; $i -lt 15; $i++) {
  Start-Sleep -Seconds 5
  try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing; $ok = $true; break } catch {}
}
if (-not $ok) { Write-Host "!! /health 未就绪" -ForegroundColor Red; exit 1 }
"health=$($h.status)"
$after = (Get-Process -Name python -ErrorAction SilentlyContinue | Sort-Object StartTime -Descending | Select-Object -First 1).StartTime
"proc_before=$before"
"proc_after =$after"
if ($after -le $before) { Write-Host "!! 进程新建时间未更新" -ForegroundColor Red; exit 1 }

"--- 4) 本机 8001 服务内容标记（确认静态文件已生效）---"
$local = (Invoke-WebRequest -Uri "http://127.0.0.1:8001/m/gd.html" -UseBasicParsing -TimeoutSec 20).Content
"local_judgment=$($local.Contains('判定依据'))  local_star=$($local.Contains('评分最高'))  local_no_king=$(-not $local.Contains('王者'))"
if (-not ($local.Contains('判定依据') -and $local.Contains('评分最高') -and (-not $local.Contains('王者')))) {
  Write-Host "!! 本机服务页面内容不对" -ForegroundColor Red; exit 1
}

"--- 5) 公网页面内容标记 ---"
$pub = (Invoke-WebRequest -Uri "https://a.ai24x.com/m/gd.html" -UseBasicParsing -TimeoutSec 20).Content
"pub_judgment=$($pub.Contains('判定依据'))  pub_star=$($pub.Contains('评分最高'))  pub_no_king=$(-not $pub.Contains('王者'))"
if (-not ($pub.Contains('判定依据') -and $pub.Contains('评分最高') -and (-not $pub.Contains('王者')))) {
  Write-Host "!! 公网页面内容不对（注意浏览器缓存需 Ctrl+F5，本步骤是服务器侧直读）" -ForegroundColor Red; exit 1

Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
```
