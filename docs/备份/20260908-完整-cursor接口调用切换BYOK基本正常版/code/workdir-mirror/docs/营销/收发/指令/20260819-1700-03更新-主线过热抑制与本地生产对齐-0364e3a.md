## 背景
1. **主线过热抑制（防追高）**：板块已过热（成分股 60 日位置中位数≥85%，或 位置≥60% 且 5 日涨幅中位数≥25%）且当日平均涨幅<0 → 硬性降级为「观察」（资金/情绪逃生口不再续命），防止「涨了一波今天就大跌」的板块仍被推荐为主线。位置<60% 的大涨视为底部刚启动，不算过热。
2. **判定小字段**：成分股评分新增 chg5/pos60（用已有 K 线零上游成本计算），主线判定依据小字展示 5 日涨幅/60 日位置，过热主线标注「⚠ 高位过热」（桌面 gd.html + 手机 m/gd.html 同步）。
3. **本地/生产不一致根因**：03 今日 15:12 归档由 8f5b2ab 之前的旧代码生成（主线 3 条：煤炭/创新药CXO/机器人）；15:37 部署 8f5b2ab 后未重扫，生产继续展示旧归档；本地 15:11 新代码 + 16:2x 重扫 = 主线仅「煤炭」。本指令含**离线重扫复盘 + 清今日 bj 扫描缓存 + 重启自动补扫**，部署后生产立即与本地对齐。

## 执行步骤（PowerShell，整段复制运行）
```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "0364e3ad626b"

Set-Location $REPO

"--- 0) 备份本地脏改动再还原（防止 pull 冲突）---"
$FILES = @(
  "p/a1/api/server/app/daily_report.py",
  "p/a1/web/js/bjscreener.js",
  "p/a1/web/gd.html",
  "p/a1/web/m/gd.html"
)
$BK = "C:\backup\ai24x_a\20260819_pre_0364e3a"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  if (Test-Path "$REPO\$x") {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) { Copy-Item "$REPO\$x" "$BK\$(Split-Path $x -Leaf).bak" -Force; git -C $REPO checkout -- $x }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记检查 ---"
$markers = @(
  @{ f = "p/a1/api/server/app/daily_report.py"; m = "def _bar_heat" },
  @{ f = "p/a1/api/server/app/daily_report.py"; m = "overheat" },
  @{ f = "p/a1/web/js/bjscreener.js";           m = "高位过热" },
  @{ f = "p/a1/web/gd.html";                    m = "bjscreener.js?v=88" },
  @{ f = "p/a1/web/m/gd.html";                  m = "高位过热" }
)
$fail = 0
foreach ($mk in $markers) {
  $f = Get-Content "$REPO\$($mk.f)" -Raw -Encoding UTF8
  $hit = $f.Contains($mk.m)
  "$($mk.f) -> $hit"
  if (-not $hit) { $fail = 1 }
}
if ($fail -ne 0) { Write-Host "!! 文件标记缺失" -ForegroundColor Red; exit 1 }

"--- 3) 离线重扫今日复盘（新口径归档：chg5/pos60/overheat；预算不足自动回退当日缓存，安全）---"
Push-Location "$REPO\p\a1\api\server"
python -m app.daily_report --force
Pop-Location
$mj = "$REPO\p\a1\调研报告\04-每日跟踪\板块主攻研判\20260819\mainlines.json"
if (Test-Path $mj) {
  $ml = Get-Content $mj -Raw -Encoding UTF8 | ConvertFrom-Json
  "archive_mainlines=$($ml.mainlines -join ',')"
  "archive_observes=$($ml.observes -join ',')"
  if ($ml.mainlines -join ',' -ne '煤炭') { Write-Host "!! 归档主线非「煤炭」，请检查" -ForegroundColor Red; exit 1 }
} else {
  Write-Host "!! 归档 mainlines.json 未生成" -ForegroundColor Red; exit 1
}
$sc = "$REPO\p\a1\daily_report\cache\20260819\sector_scores"
$scTxt = Get-Content $sc -Raw -Encoding UTF8
"sector_scores_has_chg5=$($scTxt.Contains('chg5'))"
if (-not $scTxt.Contains('chg5')) { Write-Host "!! sector_scores 无 chg5 字段" -ForegroundColor Red; exit 1 }

"--- 4) 备份并清空今日 bj 扫描缓存（重启后自动用新归档补扫 hs/kc/bj/bj_all）---"
$bjcache = "$REPO\p\a1\api\server\data\bj_scan_daily.json"
if (Test-Path $bjcache) {
  Copy-Item $bjcache "$BK\bj_scan_daily.json.bak" -Force
  Remove-Item $bjcache -Force
  "bj_scan_daily.json 已备份并清空"
} else {
  "bj_scan_daily.json 不存在（跳过）"
}

"--- 5) 重启 AI24X-a1-api (8001)：服务 PID 断言 + nssm 兜底 ---"
$beforePid = (Get-CimInstance Win32_Service -Filter "Name='$SRV'").ProcessId
"svc_pid_before=$beforePid"
Restart-Service $SRV -Force
$ok = $false
$afterPid = $beforePid
for ($i = 0; $i -lt 20; $i++) {
  Start-Sleep -Seconds 5
  try {
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
    if ($svc) { $afterPid = $svc.ProcessId }
    $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
    if ($afterPid -gt 0 -and $afterPid -ne $beforePid) { $ok = $true; break }
  } catch {}
}
"svc_pid_after=$afterPid  health_ok=$ok"
if (-not $ok) {
  Write-Host "!! Restart-Service 未真正重启（PID 未变），兜底 nssm restart" -ForegroundColor Yellow
  & nssm restart $SRV 2>$null
  if ($LASTEXITCODE -ne 0) { & "C:\Program Files\NSSM\nssm.exe" restart $SRV 2>$null }
  for ($i = 0; $i -lt 20; $i++) {
    Start-Sleep -Seconds 5
    try {
      $svc2 = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
      if ($svc2) { $afterPid = $svc2.ProcessId }
      $h2 = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
      if ($afterPid -gt 0 -and $afterPid -ne $beforePid) { $ok = $true; break }
    } catch {}
  }
}
if (-not $ok) { Write-Host "!! 服务仍未重启（before=$beforePid after=$afterPid），需人工处理" -ForegroundColor Red; exit 1 }
"health=$($h.status)"

"--- 6) 等 bj 自动补扫（重启后 auto-scan 后台重扫 4 市场，本步最多等 6 分钟，未完成也会后台继续）---"
$scanKeys = @()
for ($i = 0; $i -lt 24; $i++) {
  Start-Sleep -Seconds 15
  try {
    $bj = Get-Content "$REPO\p\a1\api\server\data\bj_scan_daily.json" -Raw -Encoding UTF8 | ConvertFrom-Json
    $scanKeys = @($bj.PSObject.Properties.Name | Where-Object { $_ -match 'scan:20260819' })
    "t=$($i*15)s scan_keys_today=$($scanKeys.Count)"
    if ($scanKeys.Count -ge 4) { break }
  } catch {
    "t=$($i*15)s bj_scan_daily.json 尚未生成"
  }
}
if ($scanKeys.Count -lt 4) {
  Write-Host "!! 自动补扫进行中（4 个市场已生成 $($scanKeys.Count) 个），后台会自动继续，可稍后重查" -ForegroundColor Yellow
} else {
  "bj 自动补扫已完成（hs/kc/bj/bj_all 均已生成）"
}

"--- 7) 公网页面内容标记（静态由 nginx 直读 web 目录，git pull 即生效）---"
$pgd = (Invoke-WebRequest -Uri "https://a.ai24x.com/gd.html?x=$(Get-Random)" -UseBasicParsing -TimeoutSec 20).Content
$pm  = (Invoke-WebRequest -Uri "https://a.ai24x.com/m/gd.html?x=$(Get-Random)" -UseBasicParsing -TimeoutSec 20).Content
"pub_gd_v88=$($pgd.Contains('bjscreener.js?v=88'))"
"pub_mobile_overheat=$($pm.Contains('高位过热'))"
if (-not ($pgd.Contains('bjscreener.js?v=88') -and $pm.Contains('高位过热'))) {
  Write-Host "!! 公网内容不对（可能缓存），请 Ctrl+F5 复查" -ForegroundColor Red; exit 1
}

Write-Host "=== 全部通过 ===" -ForegroundColor Green
```
