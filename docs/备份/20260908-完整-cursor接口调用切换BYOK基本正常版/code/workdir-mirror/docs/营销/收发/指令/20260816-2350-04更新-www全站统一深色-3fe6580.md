## 背景
- 上一轮 7ac49e7 只把 www 主页切了深色，其余页面仍蓝白。老板指令「统一深色」。
- 本轮把 www **46 个 HTML 全部切到 theme-dark**（css 引用 + body class 批量替换）；privacy/terms 补深色主题链接（原来只有 base.css）；help 卡片背景修正为深色变量；ai24x 指挥中心文案同步。
- 保留不动：command-center（纯跳转页）、token-admin（内部页，独立样式）、灯塔版 V1.02/V1.03（归档页，独立内联样式）。
- 共享资源（locales.js/app.js/shell.js）未改动，版本号维持 **v=20260818b**，无需升版。

## 注意事项
- 本轮只动 www 静态 `web/` 目录，**不动 markets**，无需同步 C:\sites\markets.ai24x.com。
- 静态由 core 直接读 `C:\ai24x01\web`，git pull 即生效；仍按惯例重启 core(8002)。
- 公网验收用 `?x=w3fe6580` 缓存穿透参数。
- PowerShell 断言一律用单引号字符串，避免 `$` 被当变量。
- HEAD 校验用「目标提交是 HEAD 祖先 + 关键路径 diff 与目标一致」（指令文档自身也是新提交，不能用精确 HEAD 前缀比对）。

## 副脑04 执行脚本（整段复制运行）
```powershell
# === AI24X 更新部署验收 · 目标 3fe6580 ===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-core"
$EXP  = "3fe6580"

Set-Location $REPO

"--- 1) web 目录如有本地改动：备份后 checkout ---"
$dirtyWeb = git status --short -- web
if ($dirtyWeb) {
    Copy-Item "$REPO\web" "C:\Users\Administrator\ops\web-dirty-20260816-2350" -Recurse -Force
    git checkout -- web
    if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败" -ForegroundColor Red; exit 1 }
    "dirty backup + checkout: web"
} else {
    "clean: web"
}

"--- 2) 拉取并校验目标提交 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! 目标 $EXP 不在 HEAD 历史中" -ForegroundColor Red; exit 1 }
git diff --quiet $EXP HEAD -- web
if ($LASTEXITCODE -ne 0) { Write-Host "!! web 在 $EXP 之后仍有改动，停止" -ForegroundColor Red; exit 1 }
"target_verified=ok"

"--- 3) 重启 core (8002) ---"
Restart-Service $SRV -Force
$cok = $false
for ($i = 0; $i -lt 12; $i++) {
    Start-Sleep -Seconds 5
    try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:8002/health" -TimeoutSec 10 -UseBasicParsing; $cok = $true; break } catch {}
}
if (-not $cok) { Write-Host "!! core /health 未就绪" -ForegroundColor Red; exit 1 }
"core-health=$($h.status) commit=$($h.commit)"

"--- 4) 本地验收 ---"
$blueResid = @(Get-ChildItem "$REPO\web" -Recurse -Filter *.html | Select-String -Pattern 'theme-blue' -List)
"local_theme_blue_residual=$($blueResid.Count)"
if ($blueResid.Count -ne 0) { Write-Host "!! web 仍有 theme-blue 残留" -ForegroundColor Red; exit 1 }
$darkPages = @(Get-ChildItem "$REPO\web" -Recurse -Filter *.html | Select-String -Pattern 'theme-dark.css' -List)
"local_dark_pages=$($darkPages.Count)"
if ($darkPages.Count -lt 45) { Write-Host "!! 深色页面数量异常（<45）" -ForegroundColor Red; exit 1 }

$pp = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/pricing.html?x=l3fe6580" -TimeoutSec 20 -UseBasicParsing).Content
"local_pricing_dark=$($pp.Contains('theme-dark.css') -and $pp -match 'class=\"theme-dark\"')"
if (-not ($pp.Contains('theme-dark.css') -and $pp -match 'class=\"theme-dark\"')) { Write-Host "!! pricing 未深色" -ForegroundColor Red; exit 1 }
$pc = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/console.html?x=l3fe6580" -TimeoutSec 20 -UseBasicParsing).Content
"local_console_dark=$($pc.Contains('theme-dark.css') -and $pc -match 'class=\"theme-dark\"')"
$pv = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/privacy.html?x=l3fe6580" -TimeoutSec 20 -UseBasicParsing).Content
"local_privacy_dark=$($pv.Contains('theme-dark.css') -and $pv -match 'class=\"theme-dark\"')"
$pt = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/terms.html?x=l3fe6580" -TimeoutSec 20 -UseBasicParsing).Content
"local_terms_dark=$($pt.Contains('theme-dark.css') -and $pt -match 'class=\"theme-dark\"')"
$iv = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/index.html?x=l3fe6580" -TimeoutSec 20 -UseBasicParsing).Content
"local_index_v18b=$($iv.Contains('locales.js?v=20260818b') -and $iv.Contains('app.js?v=20260818b'))"

"--- 5) 公网验收 ---"
$pub = (Invoke-WebRequest -Uri "https://www.ai24x.com/pricing.html?x=w3fe6580" -TimeoutSec 25 -UseBasicParsing).Content
"pub_pricing_dark=$($pub.Contains('theme-dark.css') -and $pub -match 'class=\"theme-dark\"')"
"pub_pricing_no_blue=$(-not $pub.Contains('theme-blue'))"
$pubC = (Invoke-WebRequest -Uri "https://www.ai24x.com/console.html?x=w3fe6580" -TimeoutSec 25 -UseBasicParsing).Content
"pub_console_dark=$($pubC.Contains('theme-dark.css'))"
$pubP = (Invoke-WebRequest -Uri "https://www.ai24x.com/privacy.html?x=w3fe6580" -TimeoutSec 25 -UseBasicParsing).Content
"pub_privacy_dark=$($pubP.Contains('theme-dark.css') -and $pubP -match 'class=\"theme-dark\"')"
$pubV = (Invoke-WebRequest -Uri "https://www.ai24x.com/?x=w3fe6580" -TimeoutSec 25 -UseBasicParsing).Content
"pub_index_v18b=$($pubV.Contains('locales.js?v=20260818b'))"

if ($blueResid.Count -eq 0 -and $darkPages.Count -ge 45 -and $pp.Contains('theme-dark.css') -and $pc.Contains('theme-dark.css') -and $pv.Contains('theme-dark.css') -and $pt.Contains('theme-dark.css') -and $pub.Contains('theme-dark.css') -and (-not $pub.Contains('theme-blue')) -and $pubP.Contains('theme-dark.css') -and $pubV.Contains('locales.js?v=20260818b')) {
    "--- 全部 PASS ---"
    Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
} else {
    Write-Host "!! 验收未全通过" -ForegroundColor Red; exit 1
}
```
