M5:=MA(CLOSE,5);
M10:=MA(CLOSE,10);
M14:=MA(CLOSE,14);
M20:=MA(CLOSE,20);
M28:=MA(CLOSE,28);
M57:=MA(CLOSE,57);
H60:=HHV(CLOSE,60);
L60:=LLV(CLOSE,60);
CPOS:=(CLOSE-L60)/(H60-L60);
SL28:=(M28-REF(M28,6))/ABS(REF(M28,6));
SL57:=(M57-REF(M57,6))/ABS(REF(M57,6));
BEAR:=SL28<-0.0015 AND SL57<-0.0015;
BY28YD:=REF(CLOSE,1)<REF(M28,1);
AB28TD:=CLOSE>=M28;
VM20:=MA(VOL,20);
D1:=BY28YD AND AB28TD AND VOL>=VM20*1.1;
D2:=CROSS(M10,M20) AND CLOSE>=M20 AND VOL>=VM20*1.15 AND M5>=M10 AND COUNT(CLOSE<=M20,8)>=4;
B1S:=CROSS(M5,M14) AND NOT(REF(CROSS(M5,M14),1));
B2S:=CROSS(M5,M20) AND NOT(REF(CROSS(M5,M20),1));
B3S:=CROSS(M14,M28) AND NOT(REF(CROSS(M14,M28),1));
HDI:=D1 OR D2 OR REF(D1,1) OR REF(D2,1);
HBS:=B1S OR B2S OR B3S OR REF(B1S,1) OR REF(B2S,1) OR REF(B3S,1);
SL14:=(M14-REF(M14,5))/ABS(REF(M14,5));
TUP:=SL14>0.001 AND M14>REF(M14,1);
BIND:=ABS(M5-M20)/M20<=0.05 AND ABS(M10-M20)/M20<=0.05 AND M5>M10 AND M10>M20;
VEX:=VOL>=VM20 AND (VOL>=VM20*1.2 OR REF(VOL,1)>=VM20*1.2);
DIFV:=EMA(CLOSE,12)-EMA(CLOSE,26);
DEAV:=EMA(DIFV,9);
MOK:=DIFV>REF(DIFV,1) OR CROSS(DIFV,DEAV);
POK:=CPOS<=0.75;
XG:(HDI OR HBS) AND (TUP OR BIND) AND VEX AND MOK AND POK AND NOT(BEAR);
