MA5:=MA(CLOSE,5);
MA10:=MA(CLOSE,10);
MA14:=MA(CLOSE,14);
MA20:=MA(CLOSE,20);
MA28:=MA(CLOSE,28);
MA57:=MA(CLOSE,57);
ZT:=CLOSE/REF(CLOSE,1)>=1.095 AND CLOSE=HIGH;
ZT_C:=COUNT(ZT,10);
ZT_OK:=ZT_C>=1 AND ZT_C<=2;
HI60:=HHV(CLOSE,60);
LO60:=LLV(CLOSE,60);
CPOS:=(CLOSE-LO60)/(HI60-LO60);
SL28:=(MA28-REF(MA28,6))/ABS(REF(MA28,6));
SL57:=(MA57-REF(MA57,6))/ABS(REF(MA57,6));
BEAR:=SL28<-0.0015 AND SL57<-0.0015;
B28Y:=REF(CLOSE,1)<REF(MA28,1);
A28T:=CLOSE>=MA28;
VMA20:=MA(VOL,20);
D1:=B28Y AND A28T AND VOL>=VMA20*1.1;
DC2:=CROSS(MA10,MA20) AND NOT(REF(CROSS(MA10,MA20),1));
R20:=CLOSE>=MA20 AND CLOSE<=MA20*1.03;
T20:=LOW<=MA20;
M5U:=MA5>=MA10 AND MA5>REF(MA5,1) AND MA10>REF(MA10,1);
CD20:=DC2 AND M5U AND CLOSE>=MA20;
BB20:=COUNT(CLOSE<=MA20,8)>=4;
D2:=R20 AND T20 AND CD20 AND VOL>=VMA20*1.15 AND BB20 AND CLOSE>=MA14;
C514:=CROSS(MA5,MA14);
B1:=C514 AND NOT(REF(C514,1));
C520:=CROSS(MA5,MA20);
B2:=C520 AND NOT(REF(C520,1));
C1428:=CROSS(MA14,MA28);
B3:=C1428 AND NOT(REF(C1428,1));
HDI:=D1 OR D2 OR REF(D1,1) OR REF(D2,1) OR REF(D1,2) OR REF(D2,2);
HBS:=B1 OR B2 OR B3 OR REF(B1,1) OR REF(B2,1) OR REF(B3,1) OR REF(B1,2) OR REF(B2,2) OR REF(B3,2);
DIB:=HDI OR HBS;
POK:=CPOS<=0.65;
XG: ZT_OK AND DIB AND POK AND NOT(BEAR);
