## 执行步骤（PowerShell，整段复制运行）
```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "c0fe011dfe4f"

Set-Location $REPO

"--- 0) 备份本地脏改动再还原（防 pull 冲突）---"
$FILES = @(
  "p\a1\web\gd.html",
  "p\a1\web\m\gd.html",
  "p\a1\web\js\bjscreener.js",
  "p\a1\web\css\bjscreener.css",
  "p\a1\web\radar.html",
  "p\a1\api\server\app\big_cycle.py",
  "p\a1\api\server\app\bj_screener.py",
  "p\a1\api\server\app\main.py",
  "p\a1\api\server\app\daily_report.py",
  "p\a1\api\server\app\radar_screener.py",
  "p\a1\api\server\app\scoring.py"
)
$BK = "C:\backup\ai24x_a\$(Get-Date -Format yyyyMMdd)_pre_strong_layer_$EXP"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  $full = Join-Path $REPO $x
  if (Test-Path $full) {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) {
      Copy-Item $full (Join-Path $BK ((Split-Path $x -Leaf) + ".bak")) -Force
      git -C $REPO checkout -- $x
    }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"
git pull origin master
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记 ---"
$gd = Get-Content (Join-Path $REPO "p\a1\web\gd.html") -Raw -Encoding UTF8
$okJs = $gd.Contains("bjscreener.js?v=148")
$okCss = $gd.Contains("bjscreener.css?v=104")
$okTab = $gd.Contains('data-market="mlpb"') -and ($gd -notmatch 'data-market="macd"')
"mark_js148=$okJs mark_css104=$okCss mark_no_macd_tab=$okTab"
if (-not ($okJs -and $okCss -and $okTab)) { Write-Host "!! gd.html 标记缺失" -ForegroundColor Red; exit 1 }

$bc = Join-Path $REPO "p\a1\api\server\app\big_cycle.py"
if (-not (Test-Path $bc)) { Write-Host "!! 缺少 big_cycle.py" -ForegroundColor Red; exit 1 }
$bj = Get-Content (Join-Path $REPO "p\a1\api\server\app\bj_screener.py") -Raw -Encoding UTF8
$okWatch = $bj.Contains("def _filter_watch_picks") -and $bj.Contains("def build_decision") -and $bj.Contains("strong-layer-v2")
"mark_watch=$okWatch"
if (-not $okWatch) { Write-Host "!! bj_screener 强势分层标记缺失" -ForegroundColor Red; exit 1 }

$js = Get-Content (Join-Path $REPO "p\a1\web\js\bjscreener.js") -Raw -Encoding UTF8
$okPickCard = $js.Contains("function pickCardHtml") -and $js.Contains("staleAlgoBanner")
"mark_js_fn=$okPickCard"
if (-not $okPickCard) { Write-Host "!! bjscreener.js 函数缺失" -ForegroundColor Red; exit 1 }

"--- 3) 重启 AI24X-a1-api ---"
$beforePid = (Get-CimInstance Win32_Service -Filter "Name='$SRV'").ProcessId
"svc_pid_before=$beforePid"
Restart-Service $SRV -Force
$ok = $false
$afterPid = $beforePid
for ($i = 0; $i -lt 20; $i++) {
  Start-Sleep -Seconds 5
  try {
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
    if ($svc) { $afterPid = $svc.ProcessId }
    $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
    if ($afterPid -gt 0 -and $afterPid -ne $beforePid -and $h.ok) { $ok = $true; break }
  } catch {}
}
"svc_pid_after=$afterPid health_ok=$ok"
if (-not $ok) { Write-Host "!! 服务重启/健康检查失败" -ForegroundColor Red; exit 1 }

"--- 4) 公网验收 ---"
$pub = Invoke-WebRequest -Uri "https://a.ai24x.com/gd.html" -UseBasicParsing -TimeoutSec 25
"gd_status=$($pub.StatusCode)"
if ($pub.StatusCode -ne 200) { exit 1 }
if ($pub.Content -notmatch 'bjscreener\.js\?v=148') { Write-Host "!! 公网 gd 非 js v=148" -ForegroundColor Red; exit 1 }
if ($pub.Content -notmatch 'bjscreener\.css\?v=104') { Write-Host "!! 公网 gd 非 css v=104" -ForegroundColor Red; exit 1 }

$jsPub = Invoke-WebRequest -Uri "https://a.ai24x.com/js/bjscreener.js?v=148" -UseBasicParsing -TimeoutSec 25
if ($jsPub.StatusCode -ne 200) { exit 1 }
if ($jsPub.Content -notmatch 'function pickCardHtml') { Write-Host "!! 公网 js 无 pickCardHtml" -ForegroundColor Red; exit 1 }
if ($jsPub.Content -notmatch 'staleAlgoBanner') { Write-Host "!! 公网 js 无 staleAlgoBanner" -ForegroundColor Red; exit 1 }

"--- DONE ---"
Write-Host "✅ 03更新完成｜AI复盘强势分层｜EXP=$EXP HEAD=$HEAD｜公网 gd v=148/104 + API已重启" -ForegroundColor Green
```
