$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"

$EXP  = "ed7eb2d76365"

Set-Location $REPO

"--- 0) 备份本地脏改动再还原（防止 pull 冲突）---"
$FILES = @(
  "p\a1\api\server\app\bj_screener.py",
  "p\a1\api\server\app\main.py",
  "p\a1\web\js\bjscreener.js",
  "p\a1\web\gd.html",
  "p\a1\web\m\gd.html"
)
$BK = "C:\backup\ai24x_a\$(Get-Date -Format yyyyMMdd)_pre_daboduan_$EXP"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  if (Test-Path "$REPO\$x") {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) { Copy-Item "$REPO\$x" "$BK\$(Split-Path $x -Leaf).bak" -Force; git -C $REPO checkout -- $x }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"
git pull origin master
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记检查 ---"
$f1 = Get-Content "$REPO\p\a1\api\server\app\bj_screener.py" -Raw -Encoding UTF8
$f2 = Get-Content "$REPO\p\a1\web\js\bjscreener.js" -Raw -Encoding UTF8
$f3 = Get-Content "$REPO\p\a1\web\gd.html" -Raw -Encoding UTF8
$f4 = Get-Content "$REPO\p\a1\web\m\gd.html" -Raw -Encoding UTF8
$okName = $f1.Contains('def _resolve_stock_names')
$okReplay = $f1.Contains('replay_preview')

$okV = $f3.Contains('bjscreener.js?v=139')
$okMob = $f4.Contains("mkt === 'tight' || mkt === 'macd' || mkt === 'low10'")
"mark_name=$okName mark_replay=$okReplay mark_hs=$okHsKc mark_js=$okJs mark_v139=$okV mark_mob=$okMob"
if (-not ($okName -and $okReplay -and $okHsKc -and $okJs -and $okV -and $okMob)) {
  Write-Host "!! 文件标记缺失" -ForegroundColor Red
  exit 1
}

"--- 3) 重启 AI24X-a1-api (8001) ---"
$beforePid = (Get-CimInstance Win32_Service -Filter "Name='$SRV'").ProcessId
"svc_pid_before=$beforePid"
Restart-Service $SRV -Force
$ok = $false
$afterPid = $beforePid
$h = $null
for ($i = 0; $i -lt 20; $i++) {
  Start-Sleep -Seconds 5
  try {
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
    if ($svc) { $afterPid = $svc.ProcessId }
    $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing

  } catch {}
}
"svc_pid_after=$afterPid  health_ok=$ok"

  Write-Host "!! Restart-Service 未真正重启（PID 未变），兜底 nssm restart" -ForegroundColor Yellow
  & nssm restart $SRV 2>$null
  if ($LASTEXITCODE -ne 0) { & "C:\Program Files\NSSM\nssm.exe" restart $SRV 2>$null }
  for ($i = 0; $i -lt 20; $i++) {

    try {
      $svc2 = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
      if ($svc2) { $afterPid = $svc2.ProcessId }
      $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
      if ($afterPid -gt 0 -and $afterPid -ne $beforePid) { $ok = $true; break }
    } catch {}
  }
}
if (-not $ok) { Write-Host "!! 服务仍未重启（before=$beforePid after=$afterPid），需人工处理" -ForegroundColor Red; exit 1 }
"health=$($h | ConvertTo-Json -Compress)"

"--- 4) 公网静态标记 ---"

$pubJs = (Invoke-WebRequest -Uri "https://a.ai24x.com/js/bjscreener.js?v=139" -UseBasicParsing -TimeoutSec 20).Content

"pub_v139=$okPubV pub_js=$okPubJs"
if (-not ($okPubV -and $okPubJs)) {
  Write-Host "!! 公网内容不对（nginx 静态目录可能未同步到仓库 web）" -ForegroundColor Red
  exit 1
}

Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
