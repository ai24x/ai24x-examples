## 二、执行步骤（04 复制运行）

```powershell
$ErrorActionPreference = "Stop"
$work = "C:\ai24x01"
$EXP  = "d23b51d"
Set-Location $work

Write-Host "== 1. fetch + 含目标提交 ==" -ForegroundColor Cyan
git fetch origin
git pull --ff-only origin master
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { throw "HEAD 不含目标提交 $EXP" }

Write-Host "== 2. 静态已随 pull 更新（若惯用 checkout 目录：git checkout $EXP -- web/） ==" -ForegroundColor Cyan

Write-Host "== 3. 文件标记校验 ==" -ForegroundColor Cyan
$robots = Get-Content "$work\web\robots.txt" -Raw -Encoding UTF8
if (-not $robots.Contains("Sitemap: https://www.ai24x.com/sitemap.xml")) { throw "robots 缺 sitemap" }
$sm = Get-Content "$work\web\sitemap.xml" -Raw -Encoding UTF8
if (-not $sm.Contains("blog/what-is-ai-gateway.html")) { throw "sitemap 缺 Hub" }
if ($sm.Contains("register.html")) { throw "sitemap 仍含 register（应已剔除）" }
$idx = Get-Content "$work\web\index.html" -Raw -Encoding UTF8
if (-not $idx.Contains("AI Gateway guides")) { throw "首页缺 Hub 内链区" }
$hub = Get-Content "$work\web\blog\byok-ai-gateway.html" -Raw -Encoding UTF8
if (-not $hub.Contains("hreflang")) { throw "Hub 缺 hreflang" }
if (-not $hub.Contains("application/ld+json")) { throw "Hub 缺 Article schema" }
Write-Host "文件标记通过" -ForegroundColor Green

Write-Host "== 4. nginx（手工）按上文 P0 改站点配置后：nginx -t && nginx -s reload ==" -ForegroundColor Cyan
Write-Host "   重点：壳页 301 到 open；垃圾路径 404；旧 blog 404 处理" -ForegroundColor Yellow

Write-Host "== 5. 公网验收 ==" -ForegroundColor Cyan
$r = Invoke-WebRequest -UseBasicParsing -Uri "https://www.ai24x.com/robots.txt"
if ($r.StatusCode -ne 200) { throw "robots 非 200" }
$s = Invoke-WebRequest -UseBasicParsing -Uri "https://www.ai24x.com/sitemap.xml"
if ($s.StatusCode -ne 200) { throw "sitemap 非 200" }
$www = (Invoke-WebRequest -UseBasicParsing -Uri "https://www.ai24x.com/" -Headers @{ Accept = "text/html" }).Content
if (-not $www.Contains("AI Gateway guides")) { throw "公网首页缺 Hub 区" }
$hubUrl = (Invoke-WebRequest -UseBasicParsing -Uri "https://www.ai24x.com/blog/what-is-ai-gateway.html" -Headers @{ Accept = "text/html" }).Content
if (-not $hubUrl.Contains("hreflang")) { throw "公网 Hub 缺 hreflang" }
# 壳页应 301（示例）
try {
  $docs = Invoke-WebRequest -UseBasicParsing -Uri "https://www.ai24x.com/docs.html" -MaximumRedirection 0 -ErrorAction Stop
} catch {
  if ($_.Exception.Response.StatusCode -notin 301,302,308) { Write-Host "⚠️ docs.html 未 301 到 open，请补 nginx 规则" -ForegroundColor Yellow }
}
Write-Host "公网验收通过（nginx 301 项请人工确认）" -ForegroundColor Green

Write-Host "== 6. GSC ==" -ForegroundColor Cyan
Write-Host "提交/重抓: https://www.ai24x.com/sitemap.xml"
Write-Host "编制索引: 首页 + 5 Hub + pricing"
Write-Host "✅ 完成：GSC 收录优化（EXP=$EXP）" -ForegroundColor Green
```
