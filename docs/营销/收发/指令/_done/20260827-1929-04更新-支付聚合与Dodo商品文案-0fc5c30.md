## 二、执行步骤（04 复制运行）

```powershell
$ErrorActionPreference = "Stop"
$work = "C:\ai24x01"
$EXP  = "0fc5c30"
Set-Location $work

Write-Host "== 0. 工作树自检（有脏改只还原本批路径，禁止 reset --hard） ==" -ForegroundColor Cyan
git status -sb | Select-Object -First 20

Write-Host "== 1. fetch + pull（origin 与 gitee 任一可达即可；优先 origin） ==" -ForegroundColor Cyan
git fetch origin
git pull --ff-only origin master
if ($LASTEXITCODE -ne 0) {
  git fetch gitee
  git pull --ff-only gitee master
}
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { throw "HEAD 不含目标提交 $EXP — 先确认远端已 push" }

Write-Host "== 2. 关键文件/内容标记校验（防旧缓存/漏 checkout） ==" -ForegroundColor Cyan
$must = @(
  "api\pay_dodo_products.py",
  "web\js\console.js",
  "web\console.html",
  "p\open\web\js\console.js",
  "p\open\web\console.html",
  "p\markets\web\app.html"
)
foreach ($rel in $must) {
  $p = Join-Path $work $rel
  if (-not (Test-Path $p)) { throw "缺文件: $rel" }
}
Write-Host ("齐套 " + $must.Count + " 项") -ForegroundColor Green

$checks = @(
  @{ Path = "api\pay_dodo_products.py"; Needle = "AI24X AI Gateway Credits"; Anti = "AI24X Token ·" },
  @{ Path = "web\js\console.js"; Needle = "Card payments powered by Dodo Payments" },
  @{ Path = "web\console.html"; Needle = "console.js?v=20260827a"; Anti = "console.js?v=20260826s" },
  @{ Path = "p\open\web\js\console.js"; Needle = "Card payments powered by Dodo Payments" },
  @{ Path = "p\open\web\console.html"; Needle = "console.js?v=20260827a"; Anti = "console.js?v=20260826s" },
  @{ Path = "p\markets\web\app.html"; Needle = "pay.dodoSub"; Anti = "ready.creem" }
)
foreach ($c in $checks) {
  $raw = Get-Content (Join-Path $work $c.Path) -Raw -Encoding UTF8
  if ($c.Needle -and ($raw -notlike ("*" + $c.Needle + "*"))) { throw ("缺标记 " + $c.Path + " :: " + $c.Needle) }
  if ($c.Anti -and ($raw -like ("*" + $c.Anti + "*"))) { throw ("仍含禁用串 " + $c.Path + " :: " + $c.Anti) }
}
Write-Host "内容标记通过" -ForegroundColor Green

Write-Host "== 3. 重启 AI24X-core（加载新 pay_dodo_products） ==" -ForegroundColor Cyan
Restart-Service AI24X-core -ErrorAction Stop
Start-Sleep -Seconds 8
$h = Invoke-RestMethod -UseBasicParsing -Uri "https://api.ai24x.com/health" -Headers @{ Accept = "application/json" }
"health ok commit=" + $h.commit

Write-Host "== 4. Dodo live 商品同步（直接 Python 调用；HTTP 同步在双因素下会 403） ==" -ForegroundColor Cyan
Push-Location "$work\api"
$env:PYTHONIOENCODING = "utf-8"
$syncOut = python -c "import asyncio, json; from pay_dodo_products import sync_dodo_products; r = asyncio.run(sync_dodo_products()); print(json.dumps(r, ensure_ascii=False))"
Pop-Location
$sync = $syncOut | ConvertFrom-Json
if (-not $sync) { throw "Dodo 同步无输出" }
"dodo mode=" + $sync.mode
if ($sync.mode -ne "live") { throw "Dodo 同步未命中 live 模式（检查 .env DODO_MODE）" }
if ($sync.summary.errors -gt 0) { throw ("Dodo 同步存在错误: " + (($sync.items | ConvertTo-Json -Compress))) }
"dodo summary=" + (($sync.summary | ConvertTo-Json -Compress))
$sync.items | Where-Object { $_.action -ne "unchanged" } | ForEach-Object { "  " + $_.product + " :: " + $_.plan_id + " :: " + $_.action }

Write-Host "== 5. markets 静态同步（app.html） + 重启 markets ==" -ForegroundColor Cyan
$mkDst = "C:\sites\markets.ai24x.com"
$bak = $mkDst + "\app.html.bak-20260827-" + (Get-Date -Format "HHmmss")
Copy-Item (Join-Path $mkDst "app.html") $bak
Copy-Item (Join-Path $work "p\markets\web\app.html") (Join-Path $mkDst "app.html") -Force
Restart-Service AI24X-markets-api -ErrorAction Stop
Start-Sleep -Seconds 5
"markets app.html backed up to " + $bak

Write-Host "== 6. open-api 重启（静态由 FastAPI mount，git pull 已生效，重启更稳） ==" -ForegroundColor Cyan
try {
  $svc = Get-Service AI24X-open-api -ErrorAction SilentlyContinue
  if ($svc) {
    Restart-Service AI24X-open-api -ErrorAction Stop
    Start-Sleep -Seconds 5
    Write-Host "AI24X-open-api restarted" -ForegroundColor Green
  }
} catch {
  Write-Host ("open-api 重启跳过: " + $_) -ForegroundColor Yellow
}

Write-Host "== 7. 公网验收 ==" -ForegroundColor Cyan
$wwwC = (Invoke-WebRequest -UseBasicParsing -Uri "https://www.ai24x.com/console.html" -Headers @{ Accept = "text/html" }).Content
if ($wwwC -notlike "*console.js?v=20260827a*") { throw "www console.html 版本号未更新" }
$opC = (Invoke-WebRequest -UseBasicParsing -Uri "https://open.ai24x.com/console.html" -Headers @{ Accept = "text/html" }).Content
if ($opC -notlike "*console.js?v=20260827a*") { throw "open console.html 版本号未更新" }
$mkC = (Invoke-WebRequest -UseBasicParsing -Uri "https://markets.ai24x.com/app.html" -Headers @{ Accept = "text/html" }).Content
if ($mkC -notlike "*pay.dodoSub*") { throw "markets app.html 未含 pay.dodoSub" }
if ($mkC -like "*ready.creem*") { throw "markets app.html 仍含 ready.creem" }

Write-Host "公网验收通过" -ForegroundColor Green
Write-Host ("✅ 完成 EXP=$EXP HEAD=$HEAD") -ForegroundColor Green
```
