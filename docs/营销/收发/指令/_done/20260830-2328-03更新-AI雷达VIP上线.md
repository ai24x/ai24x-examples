## 执行步骤（PowerShell，整段复制运行）
```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "8abdd42f9265"

Set-Location $REPO

"--- 0) 备份本地脏改动再还原（防止 pull 冲突）---"
$FILES = @(
  "p\a1\api\server\app\main.py",
  "p\a1\api\server\app\ths_fuyao.py",
  "p\a1\api\server\app\radar_screener.py",
  "p\a1\web\radar.html",
  "p\a1\web\js\radar.js",
  "p\a1\web\css\radar.css",
  "p\a1\web\js\shell.js"
)
$BK = "C:\backup\ai24x_a\$(Get-Date -Format yyyyMMdd)_pre_radar_$EXP"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  if (Test-Path "$REPO\$x") {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) { Copy-Item "$REPO\$x" "$BK\$(Split-Path $x -Leaf).bak" -Force; git -C $REPO checkout -- $x }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"
git pull origin master
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记检查 ---"
$radarHtml = Get-Content "$REPO\p\a1\web\radar.html" -Raw -Encoding UTF8
$radarJs   = Get-Content "$REPO\p\a1\web\js\radar.js" -Raw -Encoding UTF8
$mainPy    = Get-Content "$REPO\p\a1\api\server\app\main.py" -Raw -Encoding UTF8
$shellJs   = Get-Content "$REPO\p\a1\web\js\shell.js" -Raw -Encoding UTF8
$okPage = $radarHtml.Contains('radar.js?v=20') -and $radarHtml.Contains('radar.css?v=7')
$okJs   = $radarJs.Contains('historyFingerprint') -and $radarJs.Contains('showRadarConfirm')
$okApi  = $mainPy.Contains('/api/radar/screener/result')
$okNav  = $shellJs.Contains('AI雷达')
$okMod  = Test-Path "$REPO\p\a1\api\server\app\radar_screener.py"
"mark_page=$okPage mark_js=$okJs mark_api=$okApi mark_nav=$okNav mark_mod=$okMod"
if (-not ($okPage -and $okJs -and $okApi -and $okNav -and $okMod)) {
  Write-Host "!! 文件标记缺失" -ForegroundColor Red
  exit 1
}

"--- 3) 重启 AI24X-a1-api (8001) ---"
$beforePid = (Get-CimInstance Win32_Service -Filter "Name='$SRV'").ProcessId
"svc_pid_before=$beforePid"
Restart-Service $SRV -Force
$ok = $false
$afterPid = $beforePid
for ($i = 0; $i -lt 20; $i++) {
  Start-Sleep -Seconds 5
  try {
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
    if ($svc) { $afterPid = $svc.ProcessId }
    $null = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
    if ($afterPid -gt 0 -and $afterPid -ne $beforePid) { $ok = $true; break }
  } catch {}
}
"svc_pid_after=$afterPid  health_ok=$ok"
if (-not $ok) {
  Write-Host "!! Restart-Service 未真正重启，兜底 nssm restart" -ForegroundColor Yellow
  & nssm restart $SRV 2>$null
  if ($LASTEXITCODE -ne 0) { & "C:\Program Files\NSSM\nssm.exe" restart $SRV 2>$null }
  for ($i = 0; $i -lt 20; $i++) {
    Start-Sleep -Seconds 5
    try {
      $svc2 = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
      if ($svc2) { $afterPid = $svc2.ProcessId }
      $null = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
      if ($afterPid -gt 0 -and $afterPid -ne $beforePid) { $ok = $true; break }
    } catch {}
  }
}
if (-not $ok) { Write-Host "!! 服务重启/健康检查失败" -ForegroundColor Red; exit 1 }

"--- 4) 公网验收 ---"
$pubRadar = Invoke-WebRequest -Uri "https://a.ai24x.com/radar.html" -UseBasicParsing -TimeoutSec 20
"radar_status=$($pubRadar.StatusCode) len=$($pubRadar.Content.Length)"
if ($pubRadar.StatusCode -ne 200) { exit 1 }
if ($pubRadar.Content -notmatch 'radar\.js\?v=20') { Write-Host "!! 公网 radar.html 版本标记不符" -ForegroundColor Red; exit 1 }
if ($pubRadar.Content -notmatch 'AI雷达') { Write-Host "!! 公网 radar 页标题缺失" -ForegroundColor Red; exit 1 }

$idx = Invoke-WebRequest -Uri "https://a.ai24x.com/" -UseBasicParsing -TimeoutSec 20
"index_status=$($idx.StatusCode)"
if ($idx.Content -notmatch 'shell\.js\?v=62') { Write-Host "!! 首页 shell 版本不符" -ForegroundColor Red; exit 1 }

try {
  $r = Invoke-WebRequest -Uri "https://a.ai24x.com/api/radar/screener/result" -UseBasicParsing -TimeoutSec 15
  "radar_api_status=$($r.StatusCode) (401/403 无 token 属正常)"
} catch {
  $code = $_.Exception.Response.StatusCode.value__
  "radar_api_status=$code (401/403 无 token 属正常)"
  if ($code -ne 401 -and $code -ne 403) { Write-Host "!! radar API 异常 HTTP $code" -ForegroundColor Red; exit 1 }
}

"--- DONE ---"
Write-Host "✅ 03更新完成｜AI雷达VIP上线｜EXP=$EXP HEAD=$HEAD｜公网 radar.html + 导航 + API 路由验收通过" -ForegroundColor Green
```
