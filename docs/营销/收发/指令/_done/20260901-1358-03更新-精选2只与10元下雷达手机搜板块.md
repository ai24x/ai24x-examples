# 【03 更新】精选最多2只 + 放出10元下 + 雷达股价筛 + 手机搜板块

> 通道：司令直连 03（deploy03.ps1）｜ 目标提交 **EXP=`28d734c563e2`**
> 03 主机：123.207.199.238 · 仓库 `C:\ai24x01` · 服务 `AI24X-a1-api`（NSSM，8001）· 备份 `C:\backup\ai24x_a\`
> a1 `/health` 无 commit：验收用「文件标记 + 进程新建 + 公网内容」

## 背景
1. AI复盘各栏目精选硬顶 2 只（`pickMaxN=2`），观察池收敛；放出「10元下」页签；静态 `bjscreener.js?v=150`。
2. AI雷达新增「股价 · 10元下」条件（`radar.js?v=21`）。
3. 手机版搜索支持板块（证券等），`m.js?v=20260901a`。

## 执行步骤（PowerShell，整段复制运行）
```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "28d734c563e2"

Set-Location $REPO

"--- 0) 备份本地脏改动再还原（防 pull 冲突）---"
$FILES = @(
  "p\a1\web\gd.html",
  "p\a1\web\m\gd.html",
  "p\a1\web\m\quote.html",
  "p\a1\web\m\js\m.js",
  "p\a1\web\m\index.html",
  "p\a1\web\m\account.html",
  "p\a1\web\m\daily.html",
  "p\a1\web\m\help.html",
  "p\a1\web\m\feedback.html",
  "p\a1\web\js\bjscreener.js",
  "p\a1\web\js\radar.js",
  "p\a1\web\radar.html",
  "p\a1\api\server\app\bj_screener.py",
  "p\a1\api\server\app\main.py",
  "p\a1\api\server\app\radar_screener.py"
)
$BK = "C:\backup\ai24x_a\$(Get-Date -Format yyyyMMdd)_pre_pick2_low10_$EXP"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  $full = Join-Path $REPO $x
  if (Test-Path $full) {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) {
      Copy-Item $full (Join-Path $BK ((Split-Path $x -Leaf) + ".bak")) -Force
      git -C $REPO checkout -- $x
    }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"
git pull origin master
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记 ---"
$gd = Get-Content (Join-Path $REPO "p\a1\web\gd.html") -Raw -Encoding UTF8
$okJs = $gd.Contains("bjscreener.js?v=150")
$okLow10 = $gd.Contains('data-market="low10"')
"mark_js150=$okJs mark_low10_tab=$okLow10"
if (-not ($okJs -and $okLow10)) { Write-Host "!! gd.html 标记缺失" -ForegroundColor Red; exit 1 }

$bj = Get-Content (Join-Path $REPO "p\a1\api\server\app\bj_screener.py") -Raw -Encoding UTF8
$okPick2 = $bj.Contains('"pickMaxN": 2') -or $bj.Contains("'pickMaxN': 2")
"mark_pickMaxN2=$okPick2"
if (-not $okPick2) { Write-Host "!! bj_screener pickMaxN=2 缺失" -ForegroundColor Red; exit 1 }

$radar = Get-Content (Join-Path $REPO "p\a1\web\radar.html") -Raw -Encoding UTF8
$okRadar = $radar.Contains("radar.js?v=21") -and $radar.Contains('id="radar-price"')
"mark_radar=$okRadar"
if (-not $okRadar) { Write-Host "!! radar.html 标记缺失" -ForegroundColor Red; exit 1 }

$mjs = Get-Content (Join-Path $REPO "p\a1\web\m\js\m.js") -Raw -Encoding UTF8
$okPlate = $mjs.Contains("PLATE_ALIAS") -and $mjs.Contains("filterSuggestRows")
"mark_m_plate=$okPlate"
if (-not $okPlate) { Write-Host "!! m.js 板块搜索标记缺失" -ForegroundColor Red; exit 1 }

$mq = Get-Content (Join-Path $REPO "p\a1\web\m\quote.html") -Raw -Encoding UTF8
$okMq = $mq.Contains("m.js?v=20260901a")
"mark_quote_v=$okMq"
if (-not $okMq) { Write-Host "!! quote.html m.js 版本缺失" -ForegroundColor Red; exit 1 }

"--- 3) 重启 AI24X-a1-api ---"
$beforePid = (Get-CimInstance Win32_Service -Filter "Name='$SRV'").ProcessId
"svc_pid_before=$beforePid"
Restart-Service $SRV -Force
$ok = $false
$afterPid = $beforePid
for ($i = 0; $i -lt 20; $i++) {
  Start-Sleep -Seconds 5
  try {
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
    if ($svc) { $afterPid = $svc.ProcessId }
    $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
    if ($afterPid -gt 0 -and $afterPid -ne $beforePid -and $h.ok) { $ok = $true; break }
  } catch {}
}
"svc_pid_after=$afterPid health_ok=$ok"
if (-not $ok) { Write-Host "!! 服务重启/健康检查失败" -ForegroundColor Red; exit 1 }

"--- 4) 公网验收 ---"
$pub = Invoke-WebRequest -Uri "https://a.ai24x.com/gd.html" -UseBasicParsing -TimeoutSec 25
"gd_status=$($pub.StatusCode)"
if ($pub.StatusCode -ne 200) { exit 1 }
if ($pub.Content -notmatch 'bjscreener\.js\?v=150') { Write-Host "!! 公网 gd 非 js v=150" -ForegroundColor Red; exit 1 }
if ($pub.Content -notmatch 'data-market="low10"') { Write-Host "!! 公网 gd 缺 10元下页签" -ForegroundColor Red; exit 1 }

$jsPub = Invoke-WebRequest -Uri "https://a.ai24x.com/js/bjscreener.js?v=150" -UseBasicParsing -TimeoutSec 25
if ($jsPub.StatusCode -ne 200) { exit 1 }
if ($jsPub.Content -notmatch 'slice\(0,\s*2\)') { Write-Host "!! 公网 js 缺精选硬顶2" -ForegroundColor Red; exit 1 }

$radPub = Invoke-WebRequest -Uri "https://a.ai24x.com/radar.html" -UseBasicParsing -TimeoutSec 25
if ($radPub.StatusCode -ne 200) { exit 1 }
if ($radPub.Content -notmatch 'radar\.js\?v=21') { Write-Host "!! 公网 radar 非 v=21" -ForegroundColor Red; exit 1 }
if ($radPub.Content -notmatch 'id="radar-price"') { Write-Host "!! 公网 radar 缺股价筛" -ForegroundColor Red; exit 1 }

$mqPub = Invoke-WebRequest -Uri "https://a.ai24x.com/m/quote.html" -UseBasicParsing -TimeoutSec 25
if ($mqPub.StatusCode -ne 200) { exit 1 }
if ($mqPub.Content -notmatch 'm\.js\?v=20260901a') { Write-Host "!! 公网 m/quote 非 20260901a" -ForegroundColor Red; exit 1 }

$mjsPub = Invoke-WebRequest -Uri "https://a.ai24x.com/m/js/m.js?v=20260901a" -UseBasicParsing -TimeoutSec 25
if ($mjsPub.StatusCode -ne 200) { exit 1 }
if ($mjsPub.Content -notmatch 'PLATE_ALIAS') { Write-Host "!! 公网 m.js 缺板块别名" -ForegroundColor Red; exit 1 }

"--- 验收通过 ---"
Write-Host "OK HEAD=$HEAD EXP=$EXP pickMaxN=2 low10+radar+m-plate" -ForegroundColor Green
```
