# 执行步骤：
# 1. git pull
# 2. 验证 HEAD 前缀匹配
# 3. 验证文件标记
# 4. 验证公网页面

$ErrorActionPreference = "Stop"
$EXP = "243195688307"
cd C:\ai24x01

# Step 1: git pull
git pull origin master 2>&1
if (-not $?) { throw "git pull failed" }

# Step 2: HEAD 校验
$HEAD = git rev-parse --short=12 HEAD
if ($HEAD -notlike "$EXP*") { throw "HEAD mismatch: expected $EXP, got $HEAD" }
Write-Host "✅ HEAD=$HEAD"

# Step 3: 文件标记检查
$FILES = @(
    "p/open/web/config/locales.js",
    "p/open/web/guides/codex.html"
)
foreach ($f in $FILES) {
    if (-not (Test-Path $f)) { throw "MISSING: $f" }
}
Write-Host "✅ 文件标记 2/2"

# Step 4: 公网验收
$URLS = @(
    "https://open.ai24x.com/guides/codex.html"
)
foreach ($u in $URLS) {
    $r = Invoke-WebRequest -Uri $u -UseBasicParsing
    if ($r.StatusCode -ne 200) { throw "$u returned $($r.StatusCode)" }
}
Write-Host "✅ 公网页面 200"

# 内容验收
$html = (Invoke-WebRequest -Uri "https://open.ai24x.com/guides/codex.html" -UseBasicParsing).Content
$checks = @(
    "locales.js?v=20260906a",
    "Long sessions burn input tokens fast",
    "Bring Your Own Key (BYOK)",
    "Cost-optimized picks"
)
foreach ($c in $checks) {
    if ($html -notmatch [regex]::Escape($c)) { throw "MISSING: $c" }
}
Write-Host "✅ 内容标记 4/4"

Write-Host "`n✅ 全部验收通过"
