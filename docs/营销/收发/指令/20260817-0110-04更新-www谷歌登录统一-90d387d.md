```powershell
# === AI24X 更新部署验收 · 目标 90d387d ===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV_WWW = "AI24X-core"
$EXP  = "90d387d"

Set-Location $REPO

"--- 1) web 目录如有本地改动：备份后 checkout ---"
$dirtyWeb = git status --short -- web
if ($dirtyWeb) {
    Copy-Item "$REPO\web" "C:\Users\Administrator\ops\web-dirty-20260817-0110" -Recurse -Force
    git checkout -- web
    if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败" -ForegroundColor Red; exit 1 }
    "dirty backup + checkout: web"
} else {
    "clean: web"
}

"--- 2) 拉取并校验目标提交 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! 目标 $EXP 不在 HEAD 历史中" -ForegroundColor Red; exit 1 }
git diff --quiet $EXP HEAD -- web
if ($LASTEXITCODE -ne 0) { Write-Host "!! web 在 $EXP 之后仍有改动，停止" -ForegroundColor Red; exit 1 }
"target_verified=ok"

"--- 3) 重启 core (8002) ---"
Restart-Service $SRV_WWW -Force
$okW = $false
for ($i = 0; $i -lt 15; $i++) {
    Start-Sleep -Seconds 5
    try { $hw = Invoke-RestMethod -Uri "http://127.0.0.1:8002/health" -TimeoutSec 10 -UseBasicParsing; $okW = $true; break } catch {}
}
if (-not $okW) { Write-Host "!! core /health 未就绪" -ForegroundColor Red; exit 1 }
"core-health=$($hw.status) commit=$($hw.commit)"

"--- 4) 本地验收：register OAuth + login href 修复 + 版本号 ---"
$rg = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/register.html?x=l90d387d" -TimeoutSec 20 -UseBasicParsing).Content
"local_register_oauth_panel=$($rg.Contains('id=\"oauth-panel\"') -and $rg.Contains('id=\"btn-oauth-google\"'))"
"local_register_google_i18n=$($rg.Contains('auth.register.google'))"
"local_register_v18d=$($rg.Contains('locales.js?v=20260818d'))"
if (-not ($rg.Contains('id="oauth-panel"') -and $rg.Contains('id="btn-oauth-google"') -and $rg.Contains('auth.register.google') -and $rg.Contains('locales.js?v=20260818d'))) { Write-Host "!! register OAuth 验收失败" -ForegroundColor Red; exit 1 }

$lg = (Invoke-WebRequest -Uri "http://127.0.0.1:8002/login.html?x=l90d387d" -TimeoutSec 20 -UseBasicParsing).Content
"local_login_href_fix=$($lg.Contains('gb.href = oauthLoginUrl(''google'')') -or $lg.Contains('gb.href = oauthLoginUrl(\"google\")'))"
"local_login_v18d=$($lg.Contains('locales.js?v=20260818d'))"
if (-not ($lg.Contains('gb.href = oauthLoginUrl') -and $lg.Contains('locales.js?v=20260818d'))) { Write-Host "!! login href 修复验收失败" -ForegroundColor Red; exit 1 }

$oldVer = @(Get-ChildItem "$REPO\web" -Recurse -Filter *.html | Select-String -Pattern 'locales.js?v=20260818c' -SimpleMatch -List)
"local_old_ver_residual=$($oldVer.Count)"
if ($oldVer.Count -ne 0) { Write-Host "!! 旧版本号 20260818c 残留 $($oldVer.Count) 处" -ForegroundColor Red; exit 1 }
$newVer = @(Get-ChildItem "$REPO\web" -Recurse -Filter *.html | Select-String -Pattern 'locales.js?v=20260818d' -SimpleMatch -List)
"local_new_ver_pages=$($newVer.Count)"
if ($newVer.Count -lt 40) { Write-Host "!! 新版页面数量异常（<40）" -ForegroundColor Red; exit 1 }

"--- 5) 公网验收 + Google OAuth 冒烟 ---"
$pubP = (Invoke-WebRequest -Uri "https://www.ai24x.com/v1/auth/providers?x=w90d387d" -TimeoutSec 25 -UseBasicParsing).Content
"pub_providers=$pubP"
if ($pubP -notmatch '"google":true') { Write-Host "!! 公网 providers 未开启 google" -ForegroundColor Red; exit 1 }
$pubR = (Invoke-WebRequest -Uri "https://www.ai24x.com/register.html?x=w90d387d" -TimeoutSec 25 -UseBasicParsing).Content
"pub_register_oauth=$($pubR.Contains('id=\"oauth-panel\"') -and $pubR.Contains('btn-oauth-google'))"
"pub_register_v18d=$($pubR.Contains('locales.js?v=20260818d'))"
$pubL = (Invoke-WebRequest -Uri "https://www.ai24x.com/login.html?x=w90d387d" -TimeoutSec 25 -UseBasicParsing).Content
"pub_login_href_fix=$($pubL.Contains('gb.href = oauthLoginUrl'))"
"pub_login_v18d=$($pubL.Contains('locales.js?v=20260818d'))"
$loc = (& curl.exe -s -o NUL -w "%{redirect_url}" "https://www.ai24x.com/v1/auth/google/login?next=console.html")
"pub_google_location=$loc"
if ($loc -notlike "https://accounts.google.com/*") { Write-Host "!! Google OAuth 302 冒烟失败" -ForegroundColor Red; exit 1 }

if ($rg.Contains('id="oauth-panel"') -and $rg.Contains('auth.register.google') -and $lg.Contains('gb.href = oauthLoginUrl') -and $oldVer.Count -eq 0 -and $newVer.Count -ge 40 -and $pubP -match '"google":true' -and $pubR.Contains('id="oauth-panel"') -and $pubR.Contains('locales.js?v=20260818d') -and $pubL.Contains('gb.href = oauthLoginUrl') -and $loc -like "https://accounts.google.com/*") {
    "--- 全部 PASS ---"
    Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
} else {
    Write-Host "!! 验收未全通过" -ForegroundColor Red; exit 1
}
```
