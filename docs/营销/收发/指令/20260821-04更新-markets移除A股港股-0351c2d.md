```powershell
# === AI24X 更新部署验收 · 目标 0351c2d（markets 移除 A股/港股）===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV_MK = "AI24X-markets-api"
$EXP = "0351c2d"
$F = "p/markets/api/server/app/providers_us.py"

Set-Location $REPO

"--- 0) 该文件如有本地改动：备份后 checkout ---"
$dirty = git status --short -- $F
if ($dirty) {
    Copy-Item "$REPO\$F" "C:\Users\Administrator\ops\dirty-providers_us-20260821-cn" -Force
    git checkout -- $F
    if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败" -ForegroundColor Red; exit 1 }
    "dirty 已备份并 checkout"
} else { "clean" }

"--- 1) 拉取并校验目标提交 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! 目标 $EXP 不在 HEAD 历史中" -ForegroundColor Red; exit 1 }
git diff --quiet $EXP HEAD -- p/markets/api/server/app/providers_us.py
if ($LASTEXITCODE -ne 0) { Write-Host "!! providers_us.py 在 $EXP 之后仍有改动" -ForegroundColor Red; exit 1 }
"target_verified=ok"

"--- 2) 源码断言（无中国代码残留）---"
$src = Get-Content "$REPO\$F" -Raw -Encoding UTF8
$hasCn = ($src.Contains("_cn_code") -or $src.Contains("hkHSI") -or $src.Contains("sh000001") -or $src.Contains("sz399001"))
"src_no_cn=$(-not $hasCn) src_has_auction_guard=$($src.Contains("_drop_auction_today_bar"))"
if ($hasCn) { Write-Host "!! 源码仍有中国代码残留" -ForegroundColor Red; exit 1 }

"--- 3) 重启 markets (18012) ---"
Restart-Service $SRV_MK -Force
$ok = $false
for ($i = 0; $i -lt 15; $i++) {
    Start-Sleep -Seconds 5
    try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:18012/health" -TimeoutSec 10 -UseBasicParsing; $ok = $true; break } catch {}
}
if (-not $ok) { Write-Host "!! markets /health 未就绪" -ForegroundColor Red; exit 1 }
"markets-health=$($h.status)"
if ($h.status -ne "ok") { Write-Host "!! markets 状态异常" -ForegroundColor Red; exit 1 }

"--- 4) 本地验收：A股/港股必须 not found，美股必须正常 ---"
$fail = 0
foreach ($sym in @("上证指数", "sh000001", "sz399001", "hkHSI")) {
    $enc = [uri]::EscapeDataString($sym)
    $q = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/quote?symbol=$enc" -TimeoutSec 25 -UseBasicParsing
    $okQ = ($q.code -eq -1 -and "$($q.msg)" -like "*not found*")
    if (-not $okQ) { $fail++ }
    "quote[$sym] code=$($q.code) msg=$($q.msg) ok=$okQ"
}
$k = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=sh000001&period=day&count=120" -TimeoutSec 25 -UseBasicParsing
$okK = ($k.code -eq -1)
if (-not $okK) { $fail++ }
"kline[sh000001] code=$($k.code) ok=$okK"
$a = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/quote?symbol=AAPL" -TimeoutSec 25 -UseBasicParsing
$okA = ($a.code -eq 0 -and $a.data.name -match "Apple")
if (-not $okA) { $fail++ }
"quote[AAPL] code=$($a.code) name=$($a.data.name) ok=$okA"
$k2 = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=AAPL&period=day&count=120" -TimeoutSec 25 -UseBasicParsing
$okK2 = ($k2.code -eq 0 -and $k2.data.candles.Count -ge 100)
if (-not $okK2) { $fail++ }
"kline[AAPL] code=$($k2.code) bars=$($k2.data.candles.Count) ok=$okK2"
if ($fail -gt 0) { Write-Host "!! 验收存在 FAIL x$fail" -ForegroundColor Red; exit 1 }

Write-Host "=== 部署成功 ✅ ===" -ForegroundColor Green
```
