```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"

$EXP  = "60b330c"

"--- 0) 备份本地脏改动再还原（防止 pull 冲突）---"

New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  if (Test-Path "$REPO\$x") {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) { Copy-Item "$REPO\$x" "$BK\$(Split-Path $x -Leaf).bak" -Force; git -C $REPO checkout -- $x }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"
git pull

"--- 2) 文件标记检查 ---"
$f = Get-Content "$REPO\p\a1\web\m\gd.html" -Raw -Encoding UTF8
"m_intraday_note=$($f.Contains('今日未收盘：展示上一交易日收盘'))"
"m_new_regex=$($f.Contains('_ml = String(mm[1]).match'))"
if (-not ($f.Contains('今日未收盘：展示上一交易日收盘') -and $f.Contains('_ml = String(mm[1]).match') -and $f.Contains('_ml.map(function(s){ return stripMdLink(s); }).join'))) {

"--- 3) 重启 AI24X-a1-api (8001) ---"
$before = (Get-Process -Name python -ErrorAction SilentlyContinue | Sort-Object StartTime -Descending | Select-Object -First 1).StartTime

for ($i = 0; $i -lt 15; $i++) {
  Start-Sleep -Seconds 5
  try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing; $ok = $true; break } catch {}
}
if (-not $ok) { Write-Host "!! /health 未就绪" -ForegroundColor Red; exit 1 }
"health=$($h.status)"
$after = (Get-Process -Name python -ErrorAction SilentlyContinue | Sort-Object StartTime -Descending | Select-Object -First 1).StartTime
"proc_before=$before"
"proc_after =$after"

"--- 4) 公网页面内容标记（m/gd.html 走 Nginx 静态直读，Ctrl+F5 前服务器侧已是最新）---"

"pub_new_regex=$($pub.Contains('_ml = String(mm[1]).match'))"
if (-not ($pub.Contains('今日未收盘：展示上一交易日收盘') -and $pub.Contains('_ml = String(mm[1]).match'))) {

Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
```
