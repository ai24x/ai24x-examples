```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "2da396bf9976"

Set-Location $REPO

"--- 0) 备份本地脏改动再还原（防止 pull 冲突）---"
$FILES = @(
  "p\a1\web\gd.html",
  "p\a1\web\m\gd.html",
  "p\a1\web\js\bjscreener.js",
  "p\a1\web\css\bjscreener.css",
  "p\a1\web\js\shell.js",
  "p\a1\api\server\app\bj_screener.py",
  "p\a1\api\server\app\main.py",
  "p\a1\api\server\app\daily_report.py"
)
$BK = "C:\backup\ai24x_a\$(Get-Date -Format yyyyMMdd)_pre_AIfupan10_$EXP"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  if (Test-Path (Join-Path $REPO $x)) {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) {
      Copy-Item (Join-Path $REPO $x) (Join-Path $BK ((Split-Path $x -Leaf) + ".bak")) -Force
      git -C $REPO checkout -- $x
    }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"
git pull origin master
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记检查 ---"
$f1 = Get-Content (Join-Path $REPO "p\a1\web\gd.html") -Raw -Encoding UTF8
$okTab = $f1.Contains('data-market="mlpb"')
$okRace = $f1.Contains('id="bj-raceboard"')
$okJs = $f1.Contains("bjscreener.js?v=132")
$okCss = $f1.Contains("bjscreener.css?v=98")
$okShell = $f1.Contains("shell.js?v=61")
"mark_mlpb_tab=$okTab mark_raceboard=$okRace mark_js=$okJs mark_css=$okCss mark_shell=$okShell"
if (-not ($okTab -and $okRace -and $okJs -and $okCss -and $okShell)) {
  Write-Host "!! gd.html 标记缺失" -ForegroundColor Red; exit 1
}
$f2 = Get-Content (Join-Path $REPO "p\a1\api\server\app\bj_screener.py") -Raw -Encoding UTF8
$okView = $f2.Contains("def mlpb_view")
"mark_mlpb_view=$okView"
if (-not $okView) { Write-Host "!! bj_screener.py 缺少 mlpb_view" -ForegroundColor Red; exit 1 }

"--- 3) 重启 AI24X-a1-api (8001) ---"
$beforePid = (Get-CimInstance Win32_Service -Filter "Name='$SRV'").ProcessId
"svc_pid_before=$beforePid"
Restart-Service $SRV -Force
$ok = $false
$afterPid = $beforePid
$h = $null
for ($i = 0; $i -lt 20; $i++) {
  Start-Sleep -Seconds 5
  try {
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
    if ($svc) { $afterPid = $svc.ProcessId }
    $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
    if ($afterPid -gt 0 -and $afterPid -ne $beforePid) { $ok = $true; break }
  } catch {}
}
"svc_pid_after=$afterPid health_ok=$ok"
if (-not $ok) {
  Write-Host "!! Restart-Service 未真正重启，兜底 nssm restart" -ForegroundColor Yellow
  & nssm restart $SRV 2>$null
  if ($LASTEXITCODE -ne 0) { & "C:\Program Files\NSSM\nssm.exe" restart $SRV 2>$null }
  for ($i = 0; $i -lt 20; $i++) {
    Start-Sleep -Seconds 5
    try {
      $svc2 = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
      if ($svc2) { $afterPid = $svc2.ProcessId }
      $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
      if ($afterPid -gt 0 -and $afterPid -ne $beforePid) { $ok = $true; break }
    } catch {}
  }
}
if (-not $ok) { Write-Host "!! 服务仍未重启（before=$beforePid after=$afterPid）" -ForegroundColor Red; exit 1 }
"health=$($h.status)"

"--- 4) 公网页面内容标记 ---"
$pub = (Invoke-WebRequest -Uri "https://a.ai24x.com/gd.html" -UseBasicParsing -TimeoutSec 20).Content
$pTab = $pub.Contains('data-market="mlpb"')
$pRace = $pub.Contains("bj-raceboard")
$pJs = $pub.Contains("bjscreener.js?v=132")
$pName = $pub.Contains("主线回踩低吸")
"pub_mlpb=$pTab pub_race=$pRace pub_v132=$pJs pub_name=$pName"
if (-not ($pTab -and $pRace -and $pJs -and $pName)) {
  Write-Host "!! 公网 gd.html 内容不对" -ForegroundColor Red; exit 1
}

Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
```
