## 二、执行步骤（整段复制 PowerShell）

```powershell
# === AI24X 更新部署验收 · 目标 8cb9de4a716d ===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-core"
$EXP  = "8cb9de4a716d"

$r1 = (git ls-remote origin master | Select-Object -First 1)
Start-Sleep -Seconds 1
$r2 = (git ls-remote origin master | Select-Object -First 1)
if ($r1 -ne $r2) { Write-Host "!! 远端引用变动，稍后重试" -ForegroundColor Red; exit 1 }
Set-Location $REPO
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
if ($HEAD -notlike "$EXP*") { Write-Host "!! HEAD 不匹配，期望前缀 $EXP 实际 $HEAD" -ForegroundColor Red; exit 1 }

Restart-Service $SRV -Force
$ok = $false
for ($i = 0; $i -lt 12; $i++) {
    Start-Sleep -Seconds 5
    try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:8002/health" -TimeoutSec 10 -UseBasicParsing; $ok = $true; break } catch {}
}
if (-not $ok) { Write-Host "!! /health 未就绪" -ForegroundColor Red; exit 1 }
"health=$($h.status) commit=$($h.commit)"
if ($h.commit -notlike "$EXP*") { Write-Host "!! 服务未生效，commit=$($h.commit)" -ForegroundColor Red; exit 1 }

$pub = Invoke-RestMethod -Uri "https://api.ai24x.com/health" -TimeoutSec 15 -UseBasicParsing
"public_health=$($pub.status) commit=$($pub.commit)"
if ($pub.commit -notlike "$EXP*") { Write-Host "!! 公网 commit 未生效" -ForegroundColor Red; exit 1 }

# —— 峰谷接口验收 ——
$models = Invoke-RestMethod -Uri "https://api.ai24x.com/v1/models" -TimeoutSec 15 -UseBasicParsing
"period=$($models.period)"
$flash = $models.vip_picks | Where-Object { $_.id -eq "vip-ds-flash" } | Select-Object -First 1
$pro   = $models.vip_picks | Where-Object { $_.id -eq "vip-ds-pro" } | Select-Object -First 1
"flash_peak=$($flash.peak_out_mult) pro_peak=$($pro.peak_out_mult)"
if ($models.period -ne "peak" -and $models.period -ne "offpeak") { Write-Host "!! period 字段异常" -ForegroundColor Red; exit 1 }
if ([string]$flash.peak_out_mult -ne "6" -or [string]$pro.peak_out_mult -ne "16") { Write-Host "!! 峰时倍率未生效" -ForegroundColor Red; exit 1 }

# —— 前端版本号 + 峰谷区块验收 ——
$v = (Invoke-WebRequest -Uri "https://www.ai24x.com/pricing.html" -UseBasicParsing -TimeoutSec 20).Content
$hasV = $v -match "locales\.js\?v=20260816x"
$hasPeak = $v -match 'id="peakSection"'
$vip = (Invoke-WebRequest -Uri "https://www.ai24x.com/models/vip-picks.html" -UseBasicParsing -TimeoutSec 20).Content
$hasVip = $vip -match "peakBadge|peak out ×"
$hasStrip = $vip -match 'id="peakStrip"'
"pricing_v20260816x=$hasV peak_section=$hasPeak vip_peak=$hasVip vip_strip=$hasStrip"
if (-not $hasV -or -not $hasPeak -or -not $hasVip -or -not $hasStrip) { Write-Host "!! 前端峰谷未生效（缓存？先 Ctrl+F5/无痕验证）" -ForegroundColor Red; exit 1 }
Write-Host "=== 部署成功 ===" -ForegroundColor Green
```
