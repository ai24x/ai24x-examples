```powershell
$ErrorActionPreference = "Stop"
$work = "C:\ai24x01"
$EXP  = "b6f0b02"
Set-Location $work

Write-Host "== 0. 工作树自检（有脏改只还原本批路径，禁止 reset --hard） ==" -ForegroundColor Cyan
git status -sb | Select-Object -First 20

Write-Host "== 1. fetch + pull（origin 与 gitee 任一可达即可；优先 origin） ==" -ForegroundColor Cyan
git fetch origin
git pull --ff-only origin master
if ($LASTEXITCODE -ne 0) {
  git fetch gitee
  git pull --ff-only gitee master
}
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { throw "HEAD 不含目标提交 $EXP — 先确认远端已 push" }

Write-Host "== 2. 关键文件/目录齐套校验 ==" -ForegroundColor Cyan
$must = @(
  "web\js\api.js",
  "web\js\console.js",
  "web\config\locales.js",
  "web\index.html",
  "web\console.html",
  "web\pricing.html",
  "web\login.html",
  "web\register.html",
  "web\forgot.html",
  "web\token-admin.html",
  "web\blog\what-is-ai-gateway.html",
  "web\blog\openai-compatible-api.html",
  "web\blog\one-api-multiple-models.html",
  "web\blog\byok-ai-gateway.html",
  "web\blog\ai-api-cost-basics.html",
  "web\sitemap.xml",
  "p\open\web\js\api.js",
  "p\open\web\js\console.js",
  "p\open\web\console.html",
  "p\open\web\config\locales.js",
  "p\markets\web\js\www-links.js",
  "p\markets\web\index.html",
  "api\pay_products.py",
  "api\token_pay_service.py"
)
foreach ($rel in $must) {
  $p = Join-Path $work $rel
  if (-not (Test-Path $p)) { throw "缺文件: $rel" }
}
Write-Host ("齐套 " + $must.Count + " 项") -ForegroundColor Green

Write-Host "== 3. 内容标记校验（防旧缓存/漏 checkout） ==" -ForegroundColor Cyan
$checks = @(
  @{ Path = "web\index.html"; Needle = "Spend Smarter" },
  @{ Path = "web\js\api.js"; Needle = "safeNextUrl" },
  @{ Path = "web\js\console.js"; Needle = "Buy on AI Gateway" },
  @{ Path = "web\console.html"; Needle = "Three lanes" },
  @{ Path = "web\pricing.html"; Needle = "Top up in Account" },
  @{ Path = "web\token-admin.html"; Anti = "iamlei888" },
  @{ Path = "api\pay_products.py"; Needle = "AI Gateway credits" },
  @{ Path = "api\pay_products.py"; Anti = "PayPal or Creem" },
  @{ Path = "p\open\web\console.html"; Needle = "bring your own keys" },
  @{ Path = "p\open\web\console.html"; Anti = "BYOK_ENABLED" },
  @{ Path = "p\markets\web\js\www-links.js"; Needle = "ai24x_local_products" },
  @{ Path = "web\blog\what-is-ai-gateway.html"; Needle = "What is an AI Gateway" },
  @{ Path = "web\sitemap.xml"; Needle = "blog/what-is-ai-gateway.html" }
)
foreach ($c in $checks) {
  $raw = Get-Content (Join-Path $work $c.Path) -Raw -Encoding UTF8
  if ($c.Needle -and ($raw -notlike ("*" + $c.Needle + "*"))) { throw ("缺标记 " + $c.Path + " :: " + $c.Needle) }
  if ($c.Anti -and ($raw -like ("*" + $c.Anti + "*"))) { throw ("仍含禁用串 " + $c.Path + " :: " + $c.Anti) }
}
Write-Host "内容标记通过" -ForegroundColor Green

Write-Host "== 4. 重启 AI24X-core（加载 pay_products / token_pay_service） ==" -ForegroundColor Cyan
Restart-Service AI24X-core -ErrorAction Stop
Start-Sleep -Seconds 8
$h = Invoke-RestMethod -UseBasicParsing -Uri "https://api.ai24x.com/health" -Headers @{ Accept = "application/json" }
"health ok commit=" + $h.commit

Write-Host "== 5. open-api（若独立 NSSM；静态改动可不重启，有进程则重启更稳） ==" -ForegroundColor Cyan
try {
  $svc = Get-Service AI24X-open-api -ErrorAction SilentlyContinue
  if ($svc) {
    Restart-Service AI24X-open-api -ErrorAction Stop
    Start-Sleep -Seconds 5
    Write-Host "AI24X-open-api restarted" -ForegroundColor Green
  } else {
    Write-Host "无 AI24X-open-api 服务名，跳过（open 若挂 core/Nginx 即可）" -ForegroundColor Yellow
  }
} catch {
  Write-Host ("open-api 重启跳过: " + $_) -ForegroundColor Yellow
}

Write-Host "== 6. 公网验收 ==" -ForegroundColor Cyan
$www = (Invoke-WebRequest -UseBasicParsing -Uri "https://www.ai24x.com/" -Headers @{ Accept = "text/html" }).Content
if ($www -notlike "*Spend Smarter*") { throw "公网 www Hero 未更新（Ctrl+F5 / CDN？）" }
if ($www -like "*Private / Local AI*") { throw "www 仍露出 Private/Local AI 芯片" }

$billApi = Invoke-RestMethod -UseBasicParsing -Uri "https://api.ai24x.com/v1/billing/products" -Headers @{ Accept = "application/json" }
$titles = @($billApi.products | ForEach-Object { $_.title })
if (-not ($titles -join " " | Select-String -Pattern "Gateway credits|AI Gateway credits")) {
  # 字段可能在 title
  $blob = ($billApi | ConvertTo-Json -Depth 6)
  if ($blob -notlike "*AI Gateway credits*") { throw "billing/products 未出现 AI Gateway credits（core 未吃到新代码？）" }
}
if (($billApi | ConvertTo-Json -Depth 8) -like "*PayPal or Creem*") { throw "products 仍含 Creem 文案" }

$hub = (Invoke-WebRequest -UseBasicParsing -Uri "https://www.ai24x.com/blog/what-is-ai-gateway.html" -Headers @{ Accept = "text/html" }).Content
if ($hub -notlike "*AI Gateway*") { throw "Hub 文未部署" }

$op = (Invoke-WebRequest -UseBasicParsing -Uri "https://open.ai24x.com/console.html" -Headers @{ Accept = "text/html" }).Content
if ($op -like "*BYOK_ENABLED*") { throw "open console 仍含 BYOK_ENABLED" }

$mk = (Invoke-WebRequest -UseBasicParsing -Uri "https://markets.ai24x.com/" -Headers @{ Accept = "text/html" }).Content
if ($mk -notlike "*www-links.js*") { throw "markets 首页未挂 www-links.js" }

Write-Host "公网验收通过" -ForegroundColor Green
Write-Host ("✅ 完成 EXP=$EXP HEAD=$HEAD") -ForegroundColor Green
```
