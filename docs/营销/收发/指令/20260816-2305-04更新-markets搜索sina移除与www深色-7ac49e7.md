```powershell
# === AI24X 更新部署验收 · 目标 7ac49e7 ===
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-core"
$SRV_MARKETS = "AI24X-markets-api"
$EXP  = "7ac49e7"
$SITES = "C:\sites\markets.ai24x.com"
$KEYFILES = @(
  "p/markets/api/server/app/providers_us.py",
  "p/markets/web/app.html",
  "p/markets/web/index.html",
  "web/index.html",
  "web/config/locales.js",
  "web/js/app.js",
  "web/js/shell.js",
  "web/pricing.html"
)

Set-Location $REPO

"--- 1) 关键路径如有本地改动：备份后 checkout ---"
foreach ($f in $KEYFILES) {
    $dirty = git status --short -- $f
    if ($dirty) {
        $bn = ($f -replace "/", "_")
        Copy-Item "$REPO\$f" "C:\Users\Administrator\ops\dirty-$bn-20260816-2305" -Force
        git checkout -- $f
        if ($LASTEXITCODE -ne 0) { Write-Host "!! checkout 失败: $f" -ForegroundColor Red; exit 1 }
        "dirty backup + checkout: $f"
    } else {
        "clean: $f"
    }
}

"--- 2) 拉取并校验目标提交已包含且关键路径与目标一致 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! 目标 $EXP 不在 HEAD 历史中" -ForegroundColor Red; exit 1 }
foreach ($f in $KEYFILES) {
    git diff --quiet $EXP HEAD -- $f
    if ($LASTEXITCODE -ne 0) { Write-Host "!! $f 在 $EXP 之后仍有改动，停止" -ForegroundColor Red; exit 1 }
}
"target_verified=ok"

"--- 3) 同步 markets 静态到 nginx 服务目录（先备份）---"
foreach ($f in @("app.html", "index.html")) {
    if (Test-Path "$SITES\$f") { Copy-Item "$SITES\$f" "$SITES\$f.bak-20260816-2305" -Force }
    Copy-Item "$REPO\p\markets\web\$f" "$SITES\$f" -Force
}

"--- 4) 重启 markets (18012) ---"
Restart-Service $SRV_MARKETS -Force
$mok = $false
for ($i = 0; $i -lt 12; $i++) {
    Start-Sleep -Seconds 5
    try { $m = Invoke-RestMethod -Uri "http://127.0.0.1:18012/health" -TimeoutSec 10 -UseBasicParsing; $mok = $true; break } catch {}
}
if (-not $mok) { Write-Host "!! markets /health 未就绪" -ForegroundColor Red; exit 1 }
"markets-health=$($m.status)"

"--- 5) 重启 core (8002) ---"
Restart-Service $SRV -Force
$cok = $false
for ($i = 0; $i -lt 12; $i++) {
    Start-Sleep -Seconds 5
    try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:8002/health" -TimeoutSec 10 -UseBasicParsing; $cok = $true; break } catch {}
}
if (-not $cok) { Write-Host "!! core /health 未就绪" -ForegroundColor Red; exit 1 }
"core-health=$($h.status) commit=$($h.commit)"

"--- 6) 本地验收 ---"
$gk = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=%E5%86%A0%E7%A7%91%E7%BE%8E%E5%8D%9A" -TimeoutSec 40 -UseBasicParsing
"local_guan_kemei_candles=$($gk.data.candles.Count)"
if ($gk.data.candles.Count -ne 500) { Write-Host "!! 冠科美博非 500 根" -ForegroundColor Red; exit 1 }
$zz = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=ZZZZZZ" -TimeoutSec 40 -UseBasicParsing
"local_zz_msg=$($zz.msg)"
if ($zz.msg -match 'sina|all sources failed|circuit open') { Write-Host "!! 错误仍泄漏数据源" -ForegroundColor Red; exit 1 }
$w = Invoke-RestMethod -Uri "http://127.0.0.1:18012/api/kline?symbol=WETOUR" -TimeoutSec 40 -UseBasicParsing
"local_wetour_suggested=$($w.suggested)"
if ($w.suggested -ne "WETO") { Write-Host "!! WETOUR 建议未命中" -ForegroundColor Red; exit 1 }

$ma = Get-Content "$SITES\app.html" -Raw -Encoding UTF8
"sites_app_no_sina=$(-not $ma.Contains('Sina'))"
"sites_app_price24=$($ma.Contains('$24.9'))"
$mi = Get-Content "$SITES\index.html" -Raw -Encoding UTF8
"sites_index_amber=$($mi.Contains('#fbbf24') -and $mi.Contains('#ffd56a'))"
"sites_index_price24=$($mi.Contains('$24.9') -and $mi.Contains('$199'))"
$wi = Get-Content "$REPO\web\index.html" -Raw -Encoding UTF8
"www_index_dark=$($wi.Contains('theme-dark.css') -and $wi -match 'class="theme-dark"')"
"www_index_amber=$($wi.Contains('#fbbf24'))"
"www_index_price24=$($wi.Contains('$24.9'))"
$lc = Get-Content "$REPO\web\config\locales.js" -Raw -Encoding UTF8
"locales_price24=$($lc.Contains('$24.9') -and $lc.Contains('$199'))"
if ($lc.Contains('14.9') -or $lc.Contains('119')) { Write-Host "!! locales 仍含旧价" -ForegroundColor Red; exit 1 }

"--- 7) 公网验收 ---"
$maPub = (Invoke-WebRequest -Uri "https://markets.ai24x.com/app.html?x=s7ac49e7" -TimeoutSec 25 -UseBasicParsing).Content
"pub_app_no_sina=$(-not $maPub.Contains('Sina'))"
"pub_app_price24=$($maPub.Contains('$24.9'))"
$miPub = (Invoke-WebRequest -Uri "https://markets.ai24x.com/?x=s7ac49e7" -TimeoutSec 25 -UseBasicParsing).Content
"pub_index_price24=$($miPub.Contains('$24.9') -and $miPub.Contains('$199'))"
$gkPub = Invoke-RestMethod -Uri "https://markets.ai24x.com/api/kline?symbol=%E5%86%A0%E7%A7%91%E7%BE%8E%E5%8D%9A" -TimeoutSec 40 -UseBasicParsing
"pub_guan_kemei_candles=$($gkPub.data.candles.Count)"
$wiPub = (Invoke-WebRequest -Uri "https://www.ai24x.com/?x=w7ac49e7" -TimeoutSec 25 -UseBasicParsing).Content
"pub_www_dark=$($wiPub.Contains('theme-dark.css') -and $wiPub -match 'class="theme-dark"')"
"pub_www_v18b=$($wiPub.Contains('locales.js?v=20260818b') -and $wiPub.Contains('app.js?v=20260818b') -and $wiPub.Contains('shell.js?v=20260818b'))"
"pub_www_price24=$($wiPub.Contains('$24.9'))"
$wpPub = (Invoke-WebRequest -Uri "https://www.ai24x.com/pricing.html?x=w7ac49e7" -TimeoutSec 25 -UseBasicParsing).Content
"pub_www_pricing24=$($wpPub.Contains('$24.9') -and $wpPub.Contains('$199'))"

if ($gk.data.candles.Count -eq 500 -and $zz.msg -notmatch 'sina|all sources failed|circuit open' -and $w.suggested -eq "WETO" -and (-not $ma.Contains('Sina')) -and $mi.Contains('#fbbf24') -and $wi.Contains('theme-dark.css') -and $lc.Contains('$24.9') -and $maPub.Contains('$24.9') -and $gkPub.data.candles.Count -eq 500 -and $wiPub.Contains('theme-dark.css') -and $wiPub.Contains('locales.js?v=20260818b') -and $wiPub.Contains('$24.9') -and $wpPub.Contains('$24.9')) {
    "--- 全部 PASS ---"
    Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
} else {
    Write-Host "!! 验收未全通过" -ForegroundColor Red; exit 1
}
```
