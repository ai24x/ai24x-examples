"--- 4) 公网验收 ---"
function Probe($url, $needle) {
  try {
    $r = Invoke-WebRequest -Uri $url -UseBasicParsing -TimeoutSec 25
    $ok = if ($needle) { $r.Content.Contains($needle) } else { $true }
    "{0} -> {1} needle={2} len={3}" -f $url, $r.StatusCode, $ok, $r.Content.Length
  } catch { "{0} -> ERROR {1}" -f $url, $_.Exception.Message }
}
Probe "https://markets.ai24x.com/stocks/avgo" 'rel="canonical" href="https://markets.ai24x.com/stocks/avgo"'
Probe "https://markets.ai24x.com/stocks/coin" "Coinbase"

Probe "https://markets.ai24x.com/sitemap.xml" "stocks/avgo"
Probe "https://markets.ai24x.com/seo/index.html" "stocks/coin"

"--- 5) GSC / Bing sitemap ping ---"
curl.exe -s -o NUL -w "google_ping=%{http_code}" "https://www.google.com/ping?sitemap=https%3A%2F%2Fmarkets.ai24x.com%2Fsitemap.xml"
curl.exe -s -o NUL -w "bing_ping=%{http_code}" "https://www.bing.com/ping?sitemap=https%3A%2F%2Fmarkets.ai24x.com%2Fsitemap.xml"

Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
