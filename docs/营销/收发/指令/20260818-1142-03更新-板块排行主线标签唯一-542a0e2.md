## 背景
老板反馈「排行榜怎么又 3 个主线了」：后端把命中主线的板块都标了 mainline=true，前端渲染时给 3 行都加了「主线 ✓」标签。老板口径：**主线只有 1 个（第 1 名）**，另外 2 个是「重点关注」。
本次纯前端修复（无后端改动）：
1. 桌面 `web/js/bjscreener.js`：板块排行行内徽章 `主线 ✓` 只给 `isMain`（isTop 且 tier=king，即与复盘主线第 1 名同名的主线板块）显示；其余 mainline 板块只显示「重点关注」；状态文案「主线 · 资金+技术双确认」同步只用于 isMain。
2. 手机版 `web/m/gd.html`：同样逻辑——只有 `isTop && tier=king` 才输出「主线 ✓」+「⭐ 今日主线」，其它 mainline 只标「重点关注」。
3. `web/gd.html` 引用 `bjscreener.js?v=83 → v=84`。
本地已实测（Playwright 桌面+手机，14 项全 PASS）：板块排行 7 行中只有第 1 行含「主线 ✓ + ⭐ 今日主线」，第 2/3 行为「重点关注」且无「主线 ✓」，无 JS 报错。

```powershell
$ErrorActionPreference = "Stop"
$REPO = "C:\ai24x01"
$SRV  = "AI24X-a1-api"
$EXP  = "542a0e2"

Set-Location $REPO

"--- 0) 备份本地脏改动再还原（防止 pull 冲突）---"
$FILES = @("p\a1\web\js\bjscreener.js","p\a1\web\gd.html","p\a1\web\m\gd.html")
$BK = "C:\backup\ai24x_a\20260818_pre_542a0e2"
New-Item -ItemType Directory -Force -Path $BK | Out-Null
foreach ($x in $FILES) {
  if (Test-Path "$REPO\$x") {
    $chg = git -C $REPO status --porcelain -- $x
    if ($chg) { Copy-Item "$REPO\$x" "$BK\$(Split-Path $x -Leaf).bak" -Force; git -C $REPO checkout -- $x }
  }
}
"备份目录: $BK"

"--- 1) 拉码 + HEAD 校验 ---"
git pull
$HEAD = (git rev-parse --short=12 HEAD).Trim()
"HEAD=$HEAD"
git merge-base --is-ancestor $EXP HEAD
if ($LASTEXITCODE -ne 0) { Write-Host "!! HEAD 不含目标提交 $EXP" -ForegroundColor Red; exit 1 }

"--- 2) 文件标记检查（纪律：单引号 PS 字符串内嵌单引号用 '' 双写，勿写 \`" 转义） ---"
$js = Get-Content "$REPO\p\a1\web\js\bjscreener.js" -Raw -Encoding UTF8
$gd = Get-Content "$REPO\p\a1\web\gd.html" -Raw -Encoding UTF8
$m  = Get-Content "$REPO\p\a1\web\m\gd.html" -Raw -Encoding UTF8
$jsMain = $js.Contains('isMain ? ''<span class="tag ok">主线 ✓</span>''')
$jsSt   = $js.Contains('st = (isMain ? ''主线'' : ''重点关注'')')
$gdV    = $gd.Contains('bjscreener.js?v=84')
$mMain  = $m.Contains('if (isTop && b.tier === ''king'') t += ''<span class="tier main">主线 ✓</span>''')
"js_main=$jsMain  js_st=$jsSt  gd_v84=$gdV  m_main=$mMain"
if (-not ($jsMain -and $jsSt -and $gdV -and $mMain)) {
  Write-Host "!! 文件标记缺失" -ForegroundColor Red; exit 1
}

"--- 3) 重启 AI24X-a1-api (8001)：进程新建时间断言 + nssm 兜底 ---"
$beforePid = (Get-CimInstance Win32_Service -Filter "Name='$SRV'").ProcessId
"svc_pid_before=$beforePid"
Restart-Service $SRV -Force
$ok = $false
$afterPid = $beforePid
for ($i = 0; $i -lt 20; $i++) {
  Start-Sleep -Seconds 5
  try {
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
    if ($svc) { $afterPid = $svc.ProcessId }
    $h = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
    if ($afterPid -gt 0 -and $afterPid -ne $beforePid) { $ok = $true; break }
  } catch {}
}
"svc_pid_after=$afterPid  health_ok=$ok"
if (-not $ok) {
  Write-Host "!! Restart-Service 未真正重启（PID 未变），兜底 nssm restart" -ForegroundColor Yellow
  & nssm restart $SRV 2>$null
  if ($LASTEXITCODE -ne 0) { & "C:\Program Files\NSSM\nssm.exe" restart $SRV 2>$null }
  for ($i = 0; $i -lt 20; $i++) {
    Start-Sleep -Seconds 5
    try {
      $svc2 = Get-CimInstance Win32_Service -Filter "Name='$SRV'"
      if ($svc2) { $afterPid = $svc2.ProcessId }
      $h2 = Invoke-RestMethod -Uri "http://127.0.0.1:8001/health" -TimeoutSec 10 -UseBasicParsing
      if ($afterPid -gt 0 -and $afterPid -ne $beforePid) { $ok = $true; break }
    } catch {}
  }
}
if (-not $ok) { Write-Host "!! 服务仍未重启（before=$beforePid after=$afterPid），需人工处理" -ForegroundColor Red; exit 1 }
"health=$($h.status)"

"--- 4) 公网页面内容标记（静态变更必查） ---"
$pubGd = (Invoke-WebRequest -Uri "https://a.ai24x.com/gd.html" -UseBasicParsing -TimeoutSec 20).Content
$pubJs = (Invoke-WebRequest -Uri "https://a.ai24x.com/js/bjscreener.js?v=84" -UseBasicParsing -TimeoutSec 20).Content
$pubM  = (Invoke-WebRequest -Uri "https://a.ai24x.com/m/gd.html" -UseBasicParsing -TimeoutSec 20).Content
$pubGdV = $pubGd.Contains('bjscreener.js?v=84')
$pubJsMain = $pubJs.Contains('isMain ? ''<span class="tag ok">主线 ✓</span>''')
$pubMMain = $pubM.Contains('t += ''<span class="tier main">主线 ✓</span><span class="tier king">⭐ 今日主线</span>''')
"pub_gd_v84=$pubGdV  pub_js_main=$pubJsMain  pub_m_main=$pubMMain"
if (-not ($pubGdV -and $pubJsMain -and $pubMMain)) {
  Write-Host "!! 公网内容不对" -ForegroundColor Red; exit 1
}

Write-Host "=== 部署成功 OK ===" -ForegroundColor Green
```
