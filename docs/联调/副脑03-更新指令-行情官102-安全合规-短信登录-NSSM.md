function Set-EnvKey([string[]]$arr, [string]$key, [string]$val) {
  $prefix = $key + '='
  for ($i = 0; $i -lt $arr.Length; $i++) {
    if ($arr[$i] -match ('^\s*' + [regex]::Escape($key) + '=')) { $arr[$i] = "$prefix$val"; return ,$arr }
  }
  return ,($arr + "$prefix$val")
}

function Set-Cfg([string]$k, [string]$v) {
