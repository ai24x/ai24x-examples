## 二、执行（PowerShell · 整段复制给 OpenClaw）

```powershell
$ErrorActionPreference = 'Stop'
$PSNativeCommandUseErrorActionPreference = $false   # git 失败不直接中断（PS7.3+ Stop 会抛错），改由 $? 判断
[Console]::OutputEncoding = [System.Text.Encoding]::UTF8   # 防 git 中文输出乱码，Select-String 才能匹配
$repo = 'C:\ai24x01'

Write-Host '=== 1/3 拉码 ===' -ForegroundColor Cyan
Set-Location $repo
git checkout master
git pull origin master
if (-not $?) { Write-Host '!! 拉码失败（origin master），先别继续，回报 git pull 输出' -ForegroundColor Red; exit 1 }
git log -1 --oneline
if (-not (git log -1 --oneline | Select-String '本地身份')) { Write-Host '!! 未拉到目标提交（应含「本地身份」），先别继续，回报 git log -1' -ForegroundColor Red; exit 1 }

Write-Host '=== 2/3 重启后端（main.py + db.py + admin_ui.py + config.py 改动，必须重启 AI24X-a1-api）===' -ForegroundColor Cyan
if (Get-Service AI24X-a1-api -ErrorAction SilentlyContinue) { Restart-Service AI24X-a1-api } else { Write-Host '!! 未找到 AI24X-a1-api 服务' -ForegroundColor Red; exit 1 }
Start-Sleep -Seconds 4

Write-Host '=== 3/3 验收 ===' -ForegroundColor Cyan
try { $r = Invoke-WebRequest 'http://127.0.0.1:8001/health' -UseBasicParsing -TimeoutSec 5; Write-Host "health(8001)=$($r.StatusCode) $($r.Content)" } catch { Write-Host "health(8001) FAIL $($_.Exception.Message)" -ForegroundColor Red }
try { $r = Invoke-WebRequest 'http://127.0.0.1:18011/health' -UseBasicParsing -TimeoutSec 5; Write-Host "health(18011)=$($r.StatusCode) $($r.Content)" } catch { Write-Host "health(18011) FAIL $($_.Exception.Message)" -ForegroundColor Red }
try { $r = Invoke-WebRequest 'https://a.ai24x.com/admin20260501' -UseBasicParsing -TimeoutSec 10; $has = $r.Content.Contains('user-detail'); Write-Host "后台页面=$($r.StatusCode) 新布局类=$has（True=已生效；浏览器 Ctrl+F5）" } catch { Write-Host "后台页面 FAIL $($_.Exception.Message)" -ForegroundColor Red }
try { $r = Invoke-WebRequest 'https://a.ai24x.com/account.html' -UseBasicParsing -TimeoutSec 10; $has = $r.Content.Contains('watchscore.css?v=4'); Write-Host "account页面=$($r.StatusCode) 新资源v4=$has（True=已生效；浏览器 Ctrl+F5）" } catch { Write-Host "account页面 FAIL $($_.Exception.Message)" -ForegroundColor Red }
try { $r = Invoke-WebRequest 'https://a.ai24x.com/watchscore.html' -UseBasicParsing -TimeoutSec 10; $has = $r.Content.Contains('watchscore.css?v=4'); Write-Host "评分榜页面=$($r.StatusCode) 新资源v4=$has（True=已生效）" } catch { Write-Host "评分榜页面 FAIL $($_.Exception.Message)" -ForegroundColor Red }
Write-Host '=== 完成。回执要求：git log -1、health、后台/account/评分榜 has 结果 ===' -ForegroundColor Green
```
