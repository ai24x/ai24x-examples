## 二、执行（PowerShell · 整段复制给 OpenClaw）

```powershell
$ErrorActionPreference = 'Stop'
$PSNativeCommandUseErrorActionPreference = $false   # git 失败不直接中断（PS7.3+ Stop 会抛错），改由 $? 判断
[Console]::OutputEncoding = [System.Text.Encoding]::UTF8   # 防 git 中文输出乱码，Select-String 才能匹配
$repo = 'C:\ai24x01'

Write-Host '=== 1/3 拉码 ===' -ForegroundColor Cyan
Set-Location $repo
git checkout master
git pull origin master
if (-not $?) { Write-Host '!! 拉码失败（origin master），先别继续，回报 git pull 输出' -ForegroundColor Red; exit 1 }
git log -1 --oneline
if (-not (git log -1 --oneline | Select-String '当前账号')) { Write-Host '!! 未拉到目标提交（应含「当前账号」），先别继续，回报 git log -1' -ForegroundColor Red; exit 1 }

Write-Host '=== 2/3 本包为纯前端（web 静态文件 + docs），后端 AI24X-a1-api 无需重启 ===' -ForegroundColor Cyan
Write-Host '（若想保险重启一次也无害：Restart-Service AI24X-a1-api）' -ForegroundColor Yellow

Write-Host '=== 3/3 验收 ===' -ForegroundColor Cyan
try { $r = Invoke-WebRequest 'http://127.0.0.1:8001/health' -UseBasicParsing -TimeoutSec 5; Write-Host "health(8001)=$($r.StatusCode) $($r.Content)" } catch { Write-Host "health(8001) FAIL $($_.Exception.Message)" -ForegroundColor Red }
try { $r = Invoke-WebRequest 'https://a.ai24x.com/index.html' -UseBasicParsing -TimeoutSec 10; $h = $r.Content.Contains('.auth-card input[type="radio"]'); Write-Host "index页面=$($r.StatusCode) radio重置规则=$h（True=已生效；浏览器 Ctrl+F5）" } catch { Write-Host "index页面 FAIL $($_.Exception.Message)" -ForegroundColor Red }
try { $r = Invoke-WebRequest 'https://a.ai24x.com/account.html' -UseBasicParsing -TimeoutSec 10; $h = $r.Content.Contains('maskWlPhone'); Write-Host "account页面=$($r.StatusCode) maskWlPhone函数=$h（True=已生效）" } catch { Write-Host "account页面 FAIL $($_.Exception.Message)" -ForegroundColor Red }
try { $r = Invoke-WebRequest 'https://a.ai24x.com/js/watchscore.js' -UseBasicParsing -TimeoutSec 10; $h = $r.Content.Contains('wl-detail-col-tags'); Write-Host "watchscore.js=$($r.StatusCode) 展开对齐类=$h（True=已生效）" } catch { Write-Host "watchscore.js FAIL $($_.Exception.Message)" -ForegroundColor Red }
try { $r = Invoke-WebRequest 'https://a.ai24x.com/css/watchscore.css' -UseBasicParsing -TimeoutSec 10; $h = $r.Content.Contains('wl-detail-grid'); Write-Host "watchscore.css=$($r.StatusCode) 网格类=$h（True=已生效）" } catch { Write-Host "watchscore.css FAIL $($_.Exception.Message)" -ForegroundColor Red }
Write-Host '=== 完成。回执要求：git log -1、health、index/account/js/css has 结果 ===' -ForegroundColor Green
```
