## 一、主图指标（叠加K线）

```c
{ AI行情官 — 主图信号 }
{ ==================== MA均线系统 ==================== }
MA5:MA(CLOSE,5),COLORWHITE,LINETHICK1;
MA10:MA(CLOSE,10),COLORYELLOW,LINETHICK1;
MA14:MA(CLOSE,14),COLORMAGENTA,LINETHICK1;
MA20:MA(CLOSE,20),COLORGREEN,LINETHICK2;
MA28:MA(CLOSE,28),COLORRED,LINETHICK2;
MA57:MA(CLOSE,57),COLORBLUE,LINETHICK2;

{ ==================== 双生命线四象限着色 ==================== }
ABOVE57:=CLOSE>=MA57;
ABOVE28:=CLOSE>=MA28;
BELOW57:=CLOSE<MA57;
BELOW28:=CLOSE<MA28;
Q1_BOTH_BELOW:=BELOW57 AND BELOW28;   { 象限① 深熊 — 蓝色区 }
Q2_PRI_BELOW:=BELOW57 AND NOT(BELOW28);{ 象限② 熊反弹 — 浅蓝 }
Q3_PRI_ABOVE:=ABOVE57 AND NOT(ABOVE28);{ 象限③ 牛回调 — 浅红 }
Q4_BOTH_ABOVE:=ABOVE57 AND ABOVE28;   { 象限④ 强牛 — 红色区 }

STICKLINE(Q1_BOTH_BELOW,OPEN,CLOSE,0,0),COLORBLUE;
STICKLINE(Q2_PRI_BELOW,OPEN,CLOSE,0,0),COLORLIBLUE;
STICKLINE(Q3_PRI_ABOVE,OPEN,CLOSE,0,0),COLORLIRED;
STICKLINE(Q4_BOTH_ABOVE,OPEN,CLOSE,0,0),COLORRED;

{ ==================== 均线斜率 ==================== }
SLOPE28_6:=(MA28-REF(MA28,6))/ABS(REF(MA28,6));
SLOPE57_6:=(MA57-REF(MA57,6))/ABS(REF(MA57,6));
REGIME_BULL:=SLOPE28_6>0.0015 AND SLOPE57_6>0.0015;   { 上升趋势 }
REGIME_BEAR:=SLOPE28_6<-0.0015 AND SLOPE57_6<-0.0015;  { 下降趋势 }
REGIME_NEUTRAL:=NOT(REGIME_BULL) AND NOT(REGIME_BEAR); { 震荡 }

{ ==================== 价格位置(60日) ==================== }
HI60:=HHV(CLOSE,60);
LO60:=LLV(CLOSE,60);
CLOSEPOS:=(CLOSE-LO60)/(HI60-LO60)*100;   { 0-100, 越高越贵 }

{ ==================== 成交量 ==================== }
VMA20:=MA(VOL,20);
VOLOK:=VOL>=VMA20*0.75;   { 放量确认 }

{ ==================== 金叉/死叉检测 ==================== }
CROSS_5_10:=CROSS(MA5,MA10);      { MA5金叉MA10 }
CROSS_10_20:=CROSS(MA10,MA20);    { MA10金叉MA20 }
CROSS_5_14:=CROSS(MA5,MA14);      { MA5金叉MA14 }
CROSS_5_20:=CROSS(MA5,MA20);      { MA5金叉MA20 }
CROSS_14_28:=CROSS(MA14,MA28);    { MA14金叉MA28 }
CROSS_28_57:=CROSS(MA28,MA57);    { MA28金叉MA57 }

DCAND1:=CROSS_5_10 AND NOT(REF(CROSS_5_10,1));   { 首次金叉5/10 }
DCAND2:=CROSS_10_20 AND NOT(REF(CROSS_10_20,1)); { 首次金叉10/20 }

{ ==================== MACD ==================== }
DIF:=EMA(CLOSE,12)-EMA(CLOSE,26);
DEA:=EMA(DIF,9);
MACD_BAR:=2*(DIF-DEA),COLORSTICK;
MACD_GOLDEN:=CROSS(DIF,DEA);   { MACD金叉 }
MACD_DEAD:=CROSS(DEA,DIF);     { MACD死叉 }

{ ==================== 底信号 ==================== }
{ 底1: 放量突破MA28从下方 → 关键反转信号 }
ABOVE28_TODAY:=CLOSE>=MA28;
BELOW28_YESTERDAY:=REF(CLOSE,1)<REF(MA28,1);
VMA20_F:=MA(VOL,20);
VOLOK_BREAK:=VOL>VMA20_F*1.1;
底1: BELOW28_YESTERDAY AND ABOVE28_TODAY AND VOLOK_BREAK,COLORCYAN;

{ 底2: MA10金叉MA20附近的V型反弹 }
RECLAIM20:=CLOSE>=MA20 AND CLOSE<=MA20*1.03;
TOUCH20:=LOW<=MA20;
MA5_UP:=MA5>REF(MA5,1);
MA10_UP:=MA10>REF(MA10,1);
STUP510:=MA5>=MA10 AND MA5_UP AND MA10_UP;
CROSSDAYOK20:=DCAND2 AND STUP510 AND CLOSE>=MA20;
VOLOK2:=VOL>VMA20_F*1.15;
BASE_BELOW20:=COUNT(CLOSE<=MA20,8)>=4;
底2: RECLAIM20 AND TOUCH20 AND CROSSDAYOK20 AND VOLOK2 AND BASE_BELOW20 AND CLOSE>=MA14,COLORBLUE;

{ 小底: 深熊区的试探性底 }
HINT_BELOW14:=CLOSE<MA14;
CLOSEPOS_OK:=CLOSEPOS<=65;
NEAR_MA10:=CLOSE>=MA10*0.98 AND CLOSE<=MA10*1.03;
小底: HINT_BELOW14 AND BELOW57 AND CLOSEPOS_OK AND NEAR_MA10 AND MA5>=MA10,COLORLIGHTBLUE;

{ ==================== B系列买点 ==================== }
{ B1: MA5金叉MA14 }
B1:=CROSS_5_14 AND NOT(REF(CROSS_5_14,1)),COLORMAGENTA;

{ B2: MA5金叉MA20 }
B2:=CROSS_5_20 AND NOT(REF(CROSS_5_20,1)),COLORYELLOW;

{ B3: MA14金叉MA28 }
B3:=CROSS_14_28 AND NOT(REF(CROSS_14_28,1)),COLORRED;

{ 买2: 预测MA28即将金叉MA57(5日内) + 低位 }
JC3_SOON:=REF(CROSS_28_57,5);   { 5日后发生金叉 }
买2: JC3_SOON AND CLOSEPOS<=55,COLORWHITE;

{ ==================== 卖/险信号 ==================== }
{ 顶2: 10日内MA20/MA10多次交叉 → 顶部信号 }
CROSS_20_10:=CROSS(MA20,MA10);
CROSS_CNT_10:=COUNT(CROSS_20_10,10);
顶2: CROSS_CNT_10>=2 AND CROSS_20_10 AND CLOSEPOS>=50,COLORGREEN;

{ 卖1: MA20上穿MA10 → 中期转弱 }
卖1: CROSS_20_10 AND NOT(REF(CROSS_20_10,1)),COLORGREEN;

{ 卖2: 相对高位+MA14向MA28收敛 }
MA1_DECLINE:=MA14<REF(MA14,1) AND ABS(MA14-MA28)/MA28<0.02;
IS_HIGH:=CLOSEPOS>=60;
卖2: IS_HIGH AND MA1_DECLINE AND MA14>MA28,COLORLIGHTGREEN;

{ ==================== 风险信号 ==================== }
CROSS_10_5:=CROSS(MA10,MA5);

{ 险1: MA10下穿MA5 }
险1: CROSS_10_5 AND NOT(REF(CROSS_10_5,1)),COLORBROWN;

{ 险2: 30日内第二次MA10/MA5死叉 }
险1_COUNT_30:=COUNT(CROSS_10_5 AND NOT(REF(CROSS_10_5,1)),30);
险2: CROSS_10_5 AND 险1_COUNT_30>=2,COLORLIGHTBROWN;

{ 破位: MA10/MA5死叉+高位+双生命线上方 }
破位: CROSS_10_5 AND CLOSEPOS>=50 AND Q4_BOTH_ABOVE,COLORCYAN;

{ ==================== 绘制信号图标 ==================== }
DRAWICON(底1,LOW*0.97,1);   { 笑脸 }
DRAWICON(底2,LOW*0.97,1);
DRAWICON(小底,LOW*0.98,5);
DRAWICON(B1,LOW*0.97,5);
DRAWICON(B2,LOW*0.97,5);
DRAWICON(B3,LOW*0.97,5);
DRAWICON(买2,LOW*0.97,5);
DRAWICON(顶2,HIGH*1.03,2);
DRAWICON(卖1,HIGH*1.03,2);
DRAWICON(卖2,HIGH*1.03,2);
DRAWICON(险1,HIGH*1.03,19);
DRAWICON(险2,HIGH*1.03,19);
DRAWICON(破位,HIGH*1.03,19);

{ ==================== 文字标注 ==================== }
DRAWTEXT(底1,LOW*0.95,'底1'),COLORCYAN;
DRAWTEXT(底2,LOW*0.95,'底2'),COLORBLUE;
DRAWTEXT(小底,LOW*0.95,'小底'),COLORLIGHTBLUE;
DRAWTEXT(B1,LOW*0.95,'B1'),COLORMAGENTA;
DRAWTEXT(B2,LOW*0.95,'B2'),COLORYELLOW;
DRAWTEXT(B3,LOW*0.95,'B3'),COLORRED;
DRAWTEXT(买2,LOW*0.95,'买2'),COLORWHITE;
DRAWTEXT(顶2,HIGH*1.05,'顶2'),COLORGREEN;
DRAWTEXT(卖1,HIGH*1.05,'卖1'),COLORGREEN;
DRAWTEXT(卖2,HIGH*1.05,'卖2'),COLORLIGHTGREEN;
DRAWTEXT(险1,HIGH*1.05,'险1'),COLORBROWN;
DRAWTEXT(险2,HIGH*1.05,'险2'),COLORLIGHTBROWN;
DRAWTEXT(破位,HIGH*1.05,'破位'),COLORCYAN;

{ ==================== MACD金叉/死叉标记 ==================== }
DRAWICON(MACD_GOLDEN,LOW*0.965,1);
DRAWICON(MACD_DEAD,HIGH*1.035,2);
```

## 二、副图指标 — MACD 增强版

```c
{ AI行情官 — MACD 增强副图 }
{ 标准MACD }
DIF:EMA(CLOSE,12)-EMA(CLOSE,26),COLORWHITE;
DEA:EMA(DIF,9),COLORYELLOW;
MACD_BAR:2*(DIF-DEA),COLORSTICK;

{ 零轴 }
ZERO:0,COLORGRAY,DOTLINE;

{ 金叉死叉 }
DRAWICON(CROSS(DIF,DEA),DIF,1);
DRAWICON(CROSS(DEA,DIF),DIF,2);

{ MACD方向背景 — 金叉后粉红底，死叉后浅绿底 }
MACD_LAST_DIR:=BARSLAST(CROSS(DIF,DEA))<BARSLAST(CROSS(DEA,DIF)) OR (COUNT(CROSS(DIF,DEA),0)>0 AND COUNT(CROSS(DEA,DIF),0)=0);
STICKLINE(MACD_LAST_DIR,DIF,DEA,0.1,0),COLORFF00FF;
STICKLINE(NOT(MACD_LAST_DIR) AND COUNT(CROSS(DEA,DIF),0)>0,DIF,DEA,0.1,0),COLOR00FF00;

{ 底背离检测：价格新低，DIF不创新低 }
LLV_LOW20:=LLV(LOW,20);
LLV_DIF20:=LLV(DIF,20);
BOTTOM_DIVERGE:=LOW=LLV_LOW20 AND DIF>LLV_DIF20 AND REF(LOW,1)!=LLV_LOW20;
DRAWICON(BOTTOM_DIVERGE,DIF*0.9,1);
DRAWTEXT(BOTTOM_DIVERGE,DIF*0.8,'底背离'),COLORYELLOW;

{ 顶背离检测：价格新高，DIF不创新高 }
HHV_HIGH20:=HHV(HIGH,20);
HHV_DIF20:=HHV(DIF,20);
TOP_DIVERGE:=HIGH=HHV_HIGH20 AND DIF<HHV_DIF20 AND REF(HIGH,1)!=HHV_HIGH20;
DRAWICON(TOP_DIVERGE,DIF*1.1,2);
DRAWTEXT(TOP_DIVERGE,DIF*1.2,'顶背离'),COLORCYAN;
```

## 三、副图指标 — 量价共振

```c
{ AI行情官 — 量价共振 }
{ 成交量均线 }
VMA5:MA(VOL,5),COLORWHITE;
VMA20:MA(VOL,20),COLORYELLOW;
VMA60:MA(VOL,60),COLORMAGENTA;
VOL,VOLSTICK;

{ 放量 }
BLAST:=VOL>VMA20*1.5;
STICKLINE(BLAST,0,VOL,0,0),COLORRED;

{ 缩量 }
SHRINK:=VOL<VMA20*0.5;
STICKLINE(SHRINK,0,VOL,0,0),COLORGREEN;

{ 地量 }
GROUND_VOL:=VOL=LLV(VOL,20) AND VOL<VMA60*0.5;
DRAWTEXT(GROUND_VOL,VOL*0.5,'地量'),COLORYELLOW;

{ 量价配合判断 }
PRICE_UP:=CLOSE>REF(CLOSE,1);
PRICE_DN:=CLOSE<REF(CLOSE,1);
VOL_UP:=VOL>REF(VOL,1);
VOL_DN:=VOL<REF(VOL,1);

{ 价涨量增 = 健康 }
HEALTHY:=PRICE_UP AND VOL_UP;
DRAWTEXT(HEALTHY AND BLAST,VOL,'+'),COLORRED;

{ 价涨量缩 = 背离 }
DIVERGE_UP:=PRICE_UP AND VOL_DN AND CLOSEPOS>50;
DRAWTEXT(DIVERGE_UP,VOL,'顶背离'),COLORGREEN;

{ 价跌量缩 = 洗盘 }
WASHOUT:=PRICE_DN AND VOL_DN AND CLOSEPOS<30;
DRAWTEXT(WASHOUT,VOL,'洗盘'),COLORBLUE;
```

## 五、选股公式 — 底1信号

```c
{ AI行情官 — 底1选股 }

{ MA系统 }
MA28:=MA(CLOSE,28);
MA57:=MA(CLOSE,57);

{ 昨日在MA28下方，今日站上MA28 }
BELOW28_YESTERDAY:=REF(CLOSE,1)<REF(MA28,1);
ABOVE28_TODAY:=CLOSE>=MA28;

{ 放量确认 ≥ 20日均量1.1倍 }
VMA20:=MA(VOL,20);
VOLOK:=VOL>=VMA20*1.1;

{ 趋势不处于强烈下跌 } 
SLOPE28_6:=(MA28-REF(MA28,6))/ABS(REF(MA28,6));
SLOPE57_6:=(MA57-REF(MA57,6))/ABS(REF(MA57,6));
REGIME_BEAR:=SLOPE28_6<-0.0015 AND SLOPE57_6<-0.0015;

{ 选股条件 }
XG: BELOW28_YESTERDAY AND ABOVE28_TODAY AND VOLOK AND NOT(REGIME_BEAR);
```
