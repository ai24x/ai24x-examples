{===== 均线 =====}
MA5:=MA(C,5);
MA10:=MA(C,10);
MA20:=MA(C,20);
MA14:=MA(C,14);
MA28:=MA(C,28);
MA57:=MA(C,57);

{===== 趋势环境（Regime Filter）=====}
LB:=6;
TH:=0.0015;
R28:=(MA28-REF(MA28,LB))/REF(MA28,LB);
R57:=(MA57-REF(MA57,LB))/REF(MA57,LB);
REGUP:=R28>TH AND R57>TH;
REGDN:=R28<-TH AND R57<-TH;

ALLOW_BOTTOM_BUY:=NOT REGDN;
ALLOW_SELL_TOP:=NOT REGUP;

{===== 相对位置过滤（60根区间位置）=====}
RN:=60;
POS:=(C-LLV(C,RN))/(HHV(C,RN)-LLV(C,RN));
BOTTOM_MAX_POS:=0.55;
BUY_MAX_POS:=0.55;

{===== 底候选（金叉首日）=====}
DC1:=CROSS(MA5,MA10);   {5上穿10}
DC2:=CROSS(MA10,MA20);  {10上穿20}

{===== 小底（关注）：更早、更弱（不要求站上MA14）=====}
REC10_S:=C>=MA10 AND C<=MA10*(1+0.03);          {贴MA10收回（放宽带宽）}
BASE10:=COUNT(C<=MA10,6)>=3;                    {底部结构}
XIAODI_RAW:=BARSLAST(DC1)<=10 AND REC10_S AND BASE10 AND MA5>=MA10 AND C<MA14 AND POS<=0.70;
XIAODI:=FILTER(XIAODI_RAW,6);

{===== 底1（潜伏）：确认型，要求站上MA14 =====}
REC10:=C>=MA10 AND C<=MA10*(1+0.012);           {贴MA10收回（严格带宽）}
TOUCH10:=L<=MA10;                               {回踩触线}
BASE10B:=COUNT(C<=MA10,6)>=3;
B1_RAW:=BARSLAST(DC1)<=10 AND ( (REC10 AND (TOUCH10 OR DC1)) ) AND MA5>=MA10 AND BASE10B AND C>=MA14 AND POS<=BOTTOM_MAX_POS AND ALLOW_BOTTOM_BUY;
B1:=FILTER(B1_RAW,10);

{===== 底2（潜伏·更强确认）：10上穿20后确认，要求站上MA14 =====}
REC20:=C>=MA20 AND C<=MA20*(1+0.012);
TOUCH20:=L<=MA20;
BASE20:=COUNT(C<=MA20,8)>=4;
VOLMA20:=MA(V,20);
VOL_OK2:=V>=VOLMA20*1.15;                       {放量确认（指数若无量可能偏严格）}
B2_RAW:=BARSLAST(DC2)<=10 AND ( (REC20 AND TOUCH20) OR (DC2 AND C>=MA20) ) AND MA10>=MA20 AND BASE20 AND VOL_OK2 AND C>=MA14 AND POS<=BOTTOM_MAX_POS AND ALLOW_BOTTOM_BUY;
B2:=FILTER(B2_RAW,10);

{===== 买（看多）=====}
BUY1_RAW:=CROSS(MA14,MA28) AND ALLOW_BOTTOM_BUY;                       {趋势转强}
BUY1:=FILTER(BUY1_RAW,8);

BUY2_RAW:=BARSLAST(CROSS(MA28,MA57))=5 AND ALLOW_BOTTOM_BUY AND POS<=BUY_MAX_POS; {28上穿57后第5根}
BUY2:=FILTER(BUY2_RAW,8);

{===== 卖（看空）=====}
SELL1_RAW:=CROSS(MA20,MA5) AND ALLOW_SELL_TOP;                         {短周期转弱}
SELL1:=FILTER(SELL1_RAW,6);

SELL2_RAW:=ALLOW_SELL_TOP AND C>MA28 AND MA14<REF(MA14,1) AND ABS(MA14-MA28)/MA28<0.01;
SELL2:=FILTER(SELL2_RAW,6);

{===== 险（关注风险）=====}
RISK1_RAW:=CROSS(MA10,MA5) AND FILTER(CROSS(MA10,MA5),20);             {冷却式首次承压}
RISK1:=RISK1_RAW;

RISK2_RAW:=CROSS(MA10,MA5) AND BARSLAST(RISK1)>0 AND BARSLAST(RISK1)<=30;
RISK2:=FILTER(RISK2_RAW,10);

{===== 顶2（警惕高位）=====}
TOP2_RAW:=COUNT(CROSS(MA20,MA5),10)>=2 AND BARSLAST(CROSS(MA20,MA5))<=10 AND ALLOW_SELL_TOP;
TOP2:=FILTER(TOP2_RAW,12);

DRAWTEXT(B2,L*0.985,'底2'),COLOR38BDF8;

DRAWTEXT(BUY2,L*0.965,'买2'),COLORFF3D5C;

DRAWTEXT(RISK2,H*1.03,'险2'),COLORFFB300;
