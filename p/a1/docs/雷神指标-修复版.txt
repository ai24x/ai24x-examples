N1:=14;
N2:=28;
N3:=57;
N4:=5;
N5:=10;
N7:=20;
MA1:MA(CLOSE,N1),COLORFF8000,LINETHICK1;
MA2:MA(CLOSE,N2),COLOR0080FF,LINETHICK1;
MA3:MA(CLOSE,N3),COLOR00FF00,LINETHICK1;
MA4:=MA(CLOSE,N4);
MA5:=MA(CLOSE,N5);
MA7:=MA(CLOSE,N7);
JC1:=CROSS(MA1,MA2);
JC2:=CROSS(MA1,MA3);
JC3:=CROSS(MA2,MA3);
JC1_ONCE:=JC1 AND NOT(REF(JC1,1));
TJ3_POST:=REF(JC3,-5);
TJ3_POST_ONCE:=TJ3_POST AND NOT(REF(TJ3_POST,1));
HAS_BUY_2:=BARSLAST(TJ3_POST_ONCE)>=0;
PH:=20;
M1H:=HHV(MA1,PH);
M2H:=HHV(MA2,PH);
IS_HIGH:=MA1>MA2 AND MA1>M1H*0.9 AND MA2>M2H*0.9;
M1_PK:=REF(MA1,BARSLAST(MA1=HHV(MA1,10)));
M1_DEC:=MA1<M1_PK AND ABS(MA1-MA2)/MA2<0.01;
TS2:=HAS_BUY_2 AND IS_HIGH AND M1_DEC;
TS2_ONCE:=TS2 AND NOT(REF(TS2,1));
LSELL2:=BARSLAST(REF(TS2_ONCE,1));
TS2:=TS2 AND (LSELL2<0 OR LSELL2>5);
TS2_ONCE:=TS2 AND NOT(REF(TS2,1));
X5_20:=CROSS(MA7,MA4);
CNT520:=BARSLASTCOUNT(X5_20);
TS1:=X5_20 AND CNT520=1;
TS1_ONCE:=TS1 AND NOT(REF(TS1,1));
X5_10:=CROSS(MA5,MA4);
LR1:=BARSLAST(REF(X5_10,1));
RISK1:=X5_10 AND NOT(REF(X5_10,1)) AND (LR1<0 OR LR1>20);
LR1T:=BARSLAST(RISK1);
RISK2:=X5_10 AND LR1T>0 AND LR1T<=30;
RISK2_ONCE:=RISK2 AND NOT(REF(RISK2,1));
CNT2:=CNT520=2 AND BARSLAST(X5_20)<=10;
