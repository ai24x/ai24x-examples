MA1:MA(CLOSE,14),COLORFF8000,LINETHICK2;
MA2:MA(CLOSE,28),COLOR0080FF,LINETHICK3;
MA3:MA(CLOSE,57),COLOR00FF00,LINETHICK4;
MA4:=MA(CLOSE,5);
MA5:=MA(CLOSE,10);
MA7:=MA(CLOSE,20);
ABOVE_MA3:=CLOSE>=MA3;
JC1:=CROSS(MA1,MA2);
JC2:=CROSS(MA1,MA3);
JC3:=CROSS(MA2,MA3);
JC1_RAW:=JC1 AND NOT(REF(JC1,1));
TJ3:=REF(JC3,5);
TJ3_RAW:=TJ3 AND NOT(REF(TJ3,1));
HB2:=BARSLAST(TJ3_RAW)>=0;
PH:=20;
M1H:=HHV(MA1,PH);
M2H:=HHV(MA2,PH);
ISH:=MA1>MA2 AND MA1>M1H*0.9 AND MA2>M2H*0.9;
M1P:=REF(MA1,BARSLAST(MA1=HHV(MA1,10)));
M1D:=MA1<M1P AND ABS(MA1-MA2)/MA2<0.01;
TS2:=HB2 AND ISH AND M1D;
TS2O:=TS2 AND NOT(REF(TS2,1));
LS2:=BARSLAST(REF(TS2O,1));
TS2:=TS2 AND (LS2<0 OR LS2>5);
TS2O:=TS2 AND NOT(REF(TS2,1));
X520:=CROSS(MA7,MA4);
TS1_RAW:=X520 AND NOT(REF(X520,1));
X510:=CROSS(MA5,MA4);
LR1:=BARSLAST(REF(X510,1));
RK1_RAW:=X510 AND NOT(REF(X510,1)) AND (LR1<0 OR LR1>20);
LR1T:=BARSLAST(RK1GAP>0);
RK2_RAW:=X510 AND LR1T>0 AND LR1T<=30;
RK2O:=RK2_RAW AND NOT(REF(RK2_RAW,1));
CN2:=X520 AND REF(X520,1) AND NOT(REF(X520,2)) AND BARSLAST(X520)<=10;
RK1GAP:=BARSLAST(RK1_RAW)>15 OR BARSLAST(RK1_RAW)<0;
RK1:=RK1_RAW AND RK1GAP;
RISK_ON:=(RK1 OR RK2O OR CN2) AND ABOVE_MA3;
L_RISK:=BARSLAST(RISK_ON);
L_BELOW:=BARSLAST(NOT(ABOVE_MA3));
RISK_SUPP:=L_RISK<L_BELOW AND L_RISK>=0;
JC1GAP:=BARSLAST(JC1_RAW)>8 OR BARSLAST(JC1_RAW)<0;
JC1O:=JC1_RAW AND JC1GAP AND NOT(RISK_SUPP);
TJ3O:=TJ3_RAW AND NOT(RISK_SUPP);
TS1GAP:=BARSLAST(TS1_RAW)>15 OR BARSLAST(TS1_RAW)<0;
TS1:=TS1_RAW AND TS1GAP;
DRAWTEXT(JC1O,LOW*0.92,'雷神买1'),COLORRED;
DRAWICON(JC1O,LOW*0.9,1);
DRAWTEXT(TJ3O,LOW*0.92,'雷神买2'),COLORRED;
DRAWICON(TJ3O,LOW*0.9,1);
DRAWTEXT(TS1,HIGH*1.05,'雷神卖1'),COLORGREEN;
DRAWICON(TS1,HIGH*1.07,2);
DRAWTEXT(TS2O,HIGH*1.05,'雷神卖2'),COLORGREEN;
DRAWICON(TS2O,HIGH*1.07,2);
DRAWTEXT(RK1,HIGH*1.05,'风险1'),COLORYELLOW;
DRAWICON(RK1,HIGH*1.07,2);
DRAWTEXT(RK2O,HIGH*1.05,'风险2'),COLORYELLOW;
DRAWICON(RK2O,HIGH*1.07,2);
DRAWTEXT(CN2,HIGH*1.05,'高位风险'),COLORYELLOW;
DRAWICON(CN2,HIGH*1.07,2);
