MA28:=MA(CLOSE,28);
MA57:=MA(CLOSE,57);
B28Y:=REF(CLOSE,1)<REF(MA28,1);
A28T:=CLOSE>=MA28;
VMA20:=MA(VOL,20);
VOK:=VOL>=VMA20*1.1;
SL28:=(MA28-REF(MA28,6))/ABS(REF(MA28,6));
SL57:=(MA57-REF(MA57,6))/ABS(REF(MA57,6));
BEAR:=SL28<-0.0015 AND SL57<-0.0015;
D1_RAW:=B28Y AND A28T AND VOK;
D1GAP:=BARSLAST(D1_RAW)>8 OR BARSLAST(D1_RAW)<0;
XG: D1_RAW AND D1GAP AND NOT(BEAR);
