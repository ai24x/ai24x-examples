M5:=MA(CLOSE,5);
M10:=MA(CLOSE,10);
M14:=MA(CLOSE,14);
M20:=MA(CLOSE,20);
M28:=MA(CLOSE,28);
M57:=MA(CLOSE,57);
H60:=HHV(CLOSE,60);
L60:=LLV(CLOSE,60);
CPOS:=(CLOSE-L60)/(H60-L60);
SL28:=(M28-REF(M28,6))/ABS(REF(M28,6));
SL57:=(M57-REF(M57,6))/ABS(REF(M57,6));
BEAR:=SL28<-0.0015 AND SL57<-0.0015;
VM20:=MA(VOL,20);
UP_PCT:=(CLOSE-REF(CLOSE,1))/REF(CLOSE,1);
ZT:=UP_PCT>=0.095 AND CLOSE=HIGH;
BG:=UP_PCT>=0.07 AND VOL>=VM20*1.5;
SG:=ZT OR BG;
SG_C:=COUNT(SG,12);
SG_OK:=SG_C>=1 AND SG_C<=2;
LS:=BARSLAST(SG);
LS_OK:=LS>=2 AND LS<=12;
SH_V:=COUNT(VOL<=VM20*0.65,5)>=3;
SR:=COUNT(ABS(UP_PCT)<=0.03,5)>=4;
M20D:=M20<REF(M20,1) AND ABS(CLOSE-M20)/M20<=0.04;
M14N:=ABS(CLOSE-M14)/M14<=0.04;
SUP:=M14N OR M20D;
SGL:=REF(LOW,LS);
NBK:=CLOSE>=SGL*0.97 AND LS_OK;
SL14:=(M14-REF(M14,5))/ABS(REF(M14,5));
TUP:=SL14>0.001 AND M14>REF(M14,1);
BD:=ABS(M5-M20)/M20<=0.05 AND ABS(M10-M20)/M20<=0.05 AND M5>M10 AND M10>M20;
VEX:=VOL>=VM20 AND (VOL>=VM20*1.2 OR REF(VOL,1)>=VM20*1.2);
DIFV:=EMA(CLOSE,12)-EMA(CLOSE,26);
DEAV:=EMA(DIFV,9);
MOK:=DIFV>REF(DIFV,1) OR CROSS(DIFV,DEAV);
POK:=CPOS<=0.75;
XG: SG_OK AND LS_OK AND SH_V AND SR AND SUP AND NBK AND (TUP OR BD) AND VEX AND MOK AND POK AND NOT(BEAR);
