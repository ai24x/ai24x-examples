VMA5:MA(VOL,5),COLORWHITE;
VMA20:MA(VOL,20),COLORYELLOW;
VMA60:MA(VOL,60),COLORMAGENTA;
VOL,VOLSTICK;
HI60:=HHV(CLOSE,60);
LO60:=LLV(CLOSE,60);
CLOSEPOS:=(CLOSE-LO60)/(HI60-LO60)*100;
BL:=VOL>VMA20*1.5;
STICKLINE(BL,0,VOL,0,0),COLORRED;
SK:=VOL<VMA20*0.5;
STICKLINE(SK,0,VOL,0,0),COLORGREEN;
GV:=VOL=LLV(VOL,20) AND VOL<VMA60*0.5;
DRAWTEXT(GV,VOL*0.5,'地量'),COLORYELLOW;
PU:=CLOSE>REF(CLOSE,1);
PD:=CLOSE<REF(CLOSE,1);
VU:=VOL>REF(VOL,1);
VD:=VOL<REF(VOL,1);
HL:=PU AND VU;
DRAWTEXT(HL AND BL,VOL,'+'),COLORRED;
DV:=PU AND VD AND CLOSEPOS>50;
DRAWTEXT(DV,VOL,'顶背离'),COLORGREEN;
WO:=PD AND VD AND CLOSEPOS<30;
DRAWTEXT(WO,VOL,'洗盘'),COLORBLUE;
DRAWTEXT(FILTER(C>0,120),VOL*0.85,'AI行情官 a.ai24x.com'),COLORGRAY;
