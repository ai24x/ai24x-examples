{ ==================== MA均线系统 ==================== }
MA5:MA(CLOSE,5),COLORWHITE,LINETHICK1;
MA10:MA(CLOSE,10),COLORYELLOW,LINETHICK1;
MA14:MA(CLOSE,14),COLORMAGENTA,LINETHICK1;
MA20:MA(CLOSE,20),COLORGREEN,LINETHICK2;
MA28:MA(CLOSE,28),COLORRED,LINETHICK2;
MA57:MA(CLOSE,57),COLORBLUE,LINETHICK2;

{ ==================== 双生命线四象限着色 ==================== }
ABOVE57:=CLOSE>=MA57;
ABOVE28:=CLOSE>=MA28;
BELOW57:=CLOSE<MA57;
BELOW28:=CLOSE<MA28;
Q1_BOTH_BELOW:=BELOW57 AND BELOW28;
Q2_PRI_BELOW:=BELOW57 AND NOT(BELOW28);
Q3_PRI_ABOVE:=ABOVE57 AND NOT(ABOVE28);
Q4_BOTH_ABOVE:=ABOVE57 AND ABOVE28;

STICKLINE(Q1_BOTH_BELOW,OPEN,CLOSE,0,0),COLORBLUE;
STICKLINE(Q2_PRI_BELOW,OPEN,CLOSE,0,0),COLORFFAACC;
STICKLINE(Q3_PRI_ABOVE,OPEN,CLOSE,0,0),COLORCCAAFF;
STICKLINE(Q4_BOTH_ABOVE,OPEN,CLOSE,0,0),COLORRED;

{ ==================== 均线斜率 ==================== }
SLOPE28_6:=(MA28-REF(MA28,6))/ABS(REF(MA28,6));
SLOPE57_6:=(MA57-REF(MA57,6))/ABS(REF(MA57,6));
REGIME_BULL:=SLOPE28_6>0.0015 AND SLOPE57_6>0.0015;
REGIME_BEAR:=SLOPE28_6<-0.0015 AND SLOPE57_6<-0.0015;
REGIME_NEUTRAL:=NOT(REGIME_BULL) AND NOT(REGIME_BEAR);

{ ==================== 价格位置(60日) ==================== }
HI60:=HHV(CLOSE,60);
LO60:=LLV(CLOSE,60);
CLOSEPOS:=(CLOSE-LO60)/(HI60-LO60)*100;

{ ==================== 成交量 ==================== }
VMA20:=MA(VOL,20);
VOLOK:=VOL>=VMA20*0.75;

{ ==================== 金叉/死叉检测 ==================== }
CROSS_5_10:=CROSS(MA5,MA10);
CROSS_10_20:=CROSS(MA10,MA20);
CROSS_5_14:=CROSS(MA5,MA14);
CROSS_5_20:=CROSS(MA5,MA20);
CROSS_14_28:=CROSS(MA14,MA28);
CROSS_28_57:=CROSS(MA28,MA57);

DCAND1:=CROSS_5_10 AND NOT(REF(CROSS_5_10,1));
DCAND2:=CROSS_10_20 AND NOT(REF(CROSS_10_20,1));

{ ==================== MACD ==================== }
DIF:=EMA(CLOSE,12)-EMA(CLOSE,26);
DEA:=EMA(DIF,9);
MACD_BAR:=2*(DIF-DEA),COLORSTICK;
MACD_GOLDEN:=CROSS(DIF,DEA);
MACD_DEAD:=CROSS(DEA,DIF);

{ ==================== 底信号 ==================== }
ABOVE28_TODAY:=CLOSE>=MA28;
BELOW28_YESTERDAY:=REF(CLOSE,1)<REF(MA28,1);
VMA20_F:=MA(VOL,20);
VOLOK_BREAK:=VOL>VMA20_F*1.1;
底1: BELOW28_YESTERDAY AND ABOVE28_TODAY AND VOLOK_BREAK,COLORCYAN;

RECLAIM20:=CLOSE>=MA20 AND CLOSE<=MA20*1.03;
TOUCH20:=LOW<=MA20;
MA5_UP:=MA5>REF(MA5,1);
MA10_UP:=MA10>REF(MA10,1);
STUP510:=MA5>=MA10 AND MA5_UP AND MA10_UP;
CROSSDAYOK20:=DCAND2 AND STUP510 AND CLOSE>=MA20;
VOLOK2:=VOL>VMA20_F*1.15;
BASE_BELOW20:=COUNT(CLOSE<=MA20,8)>=4;
底2: RECLAIM20 AND TOUCH20 AND CROSSDAYOK20 AND VOLOK2 AND BASE_BELOW20 AND CLOSE>=MA14,COLORBLUE;

HINT_BELOW14:=CLOSE<MA14;
CLOSEPOS_OK:=CLOSEPOS<=65;
NEAR_MA10:=CLOSE>=MA10*0.98 AND CLOSE<=MA10*1.03;
小底: HINT_BELOW14 AND BELOW57 AND CLOSEPOS_OK AND NEAR_MA10 AND MA5>=MA10,COLORLIGHTBLUE;

{ ==================== B系列买点 ==================== }
B1:=CROSS_5_14 AND NOT(REF(CROSS_5_14,1)),COLORMAGENTA;
B2:=CROSS_5_20 AND NOT(REF(CROSS_5_20,1)),COLORYELLOW;
B3:=CROSS_14_28 AND NOT(REF(CROSS_14_28,1)),COLORRED;

JC3_SOON:=REF(CROSS_28_57,5);
买2: JC3_SOON AND CLOSEPOS<=55,COLORWHITE;

{ ==================== 卖/险信号 ==================== }
CROSS_20_10:=CROSS(MA20,MA10);
CROSS_CNT_10:=COUNT(CROSS_20_10,10);
顶2: CROSS_CNT_10>=2 AND CROSS_20_10 AND CLOSEPOS>=50,COLORGREEN;

卖1: CROSS_20_10 AND NOT(REF(CROSS_20_10,1)),COLORGREEN;

MA1_DECLINE:=MA14<REF(MA14,1) AND ABS(MA14-MA28)/MA28<0.02;
IS_HIGH:=CLOSEPOS>=60;
卖2: IS_HIGH AND MA1_DECLINE AND MA14>MA28,COLORLIGHTGREEN;

{ ==================== 风险信号 ==================== }
CROSS_10_5:=CROSS(MA10,MA5);

险1: CROSS_10_5 AND NOT(REF(CROSS_10_5,1)),COLORBROWN;

险1_COUNT_30:=COUNT(CROSS_10_5 AND NOT(REF(CROSS_10_5,1)),30);
险2: CROSS_10_5 AND 险1_COUNT_30>=2,COLORLIGHTBROWN;
