雷神出击，横扫千亿！雷神买点1或2，提示进场潜伏；风险1，尤其风险2提示，高位控好仓位；风险和雷神卖点1或2频繁出现，随时撤离战场！结合其它指标，低位看买点，高位看卖点！中线波段！

// 标记不支持的股票类型
IS_HK := CODELIKE('0%');
IS_US := (CODE >= 'A' AND CODE <= 'Z') OR (CODE >= 'a' AND CODE <= 'z');
IS_UNSUPPORTED := IS_HK OR IS_US;
// 指标参数设定
N1 := 14;
N2 := 28;
N3 := 57;
N4 := 5;
N5 := 10;
N7 := 20;
// 均线计算并显示
MA1: IF(NOT(IS_UNSUPPORTED), MA(CLOSE, N1), DRAWNULL), COLORFF8000, LINETHICK1;
MA2: IF(NOT(IS_UNSUPPORTED), MA(CLOSE, N2), DRAWNULL), COLOR0080FF, LINETHICK1;
MA3: IF(NOT(IS_UNSUPPORTED), MA(CLOSE, N3), DRAWNULL), COLOR00FF00, LINETHICK1;
// 计算但不显示5/10/20日均线
MA4 = IF(NOT(IS_UNSUPPORTED), MA(CLOSE, N4), DRAWNULL);
MA5 = IF(NOT(IS_UNSUPPORTED), MA(CLOSE, N5), DRAWNULL);
MA7 = IF(NOT(IS_UNSUPPORTED), MA(CLOSE, N7), DRAWNULL);
// 金叉条件
JC1 = IF(NOT(IS_UNSUPPORTED), CROSS(MA1, MA2), 0);
JC2 = IF(NOT(IS_UNSUPPORTED), CROSS(MA1, MA3), 0);
JC3 = IF(NOT(IS_UNSUPPORTED), CROSS(MA2, MA3), 0);
// 买点信号
JC1_ONCE = IF(NOT(IS_UNSUPPORTED), JC1 AND NOT(REF(JC1, 1)), 0);
TJ3_POST = IF(NOT(IS_UNSUPPORTED), REF(JC3, -5), 0);
TJ3_POST_ONCE = IF(NOT(IS_UNSUPPORTED), TJ3_POST AND NOT(REF(TJ3_POST, 1)), 0);
// 卖点及风险信号
HAS_BUY_POINT_2 = IF(NOT(IS_UNSUPPORTED), BARSLAST(TJ3_POST_ONCE) >= 0, 0);
PERIOD_HIGH = 20;
MA1_HIGH = IF(NOT(IS_UNSUPPORTED), HHV(MA1, PERIOD_HIGH), 0);
MA2_HIGH = IF(NOT(IS_UNSUPPORTED), HHV(MA2, PERIOD_HIGH), 0);
IS_RELATIVE_HIGH = IF(NOT(IS_UNSUPPORTED), MA1 > MA2 AND MA1 > MA1_HIGH * 0.9 AND MA2 > MA2_HIGH * 0.9, 0);
MA1_PEAK = IF(NOT(IS_UNSUPPORTED), REF(MA1, BARSLAST(MA1 = HHV(MA1, 10))), 0);
MA1_DECLINE = IF(NOT(IS_UNSUPPORTED), MA1 < MA1_PEAK AND ABS(MA1 - MA2) / MA2 < 0.01, 0);
TS2 = IF(NOT(IS_UNSUPPORTED), HAS_BUY_POINT_2 AND IS_RELATIVE_HIGH AND MA1_DECLINE, 0);
TS2_ONCE = IF(NOT(IS_UNSUPPORTED), TS2 AND NOT(REF(TS2, 1)), 0);
LAST_SELL_2 = IF(NOT(IS_UNSUPPORTED), BARSLAST(REF(TS2_ONCE, 1)), 0);
TS2 = IF(NOT(IS_UNSUPPORTED), TS2 AND (LAST_SELL_2 < 0 OR LAST_SELL_2 > 5), 0);
TS2_ONCE = IF(NOT(IS_UNSUPPORTED), TS2 AND NOT(REF(TS2, 1)), 0);
CROSS_5_20 = IF(NOT(IS_UNSUPPORTED), CROSS(MA7, MA4), 0);
COUNT_CROSS_5_20 = IF(NOT(IS_UNSUPPORTED), BARSLASTCOUNT(CROSS_5_20), 0);
TS1 = IF(NOT(IS_UNSUPPORTED), CROSS_5_20 AND COUNT_CROSS_5_20 = 1, 0);
TS1_ONCE = IF(NOT(IS_UNSUPPORTED), TS1 AND NOT(REF(TS1, 1)), 0);
CROSS_5_10 = IF(NOT(IS_UNSUPPORTED), CROSS(MA5, MA4), 0);
LAST_RISK_1 = BARSLAST(REF(CROSS_5_10, 1));
RISK_1 = IF(NOT(IS_UNSUPPORTED), CROSS_5_10 AND NOT(REF(CROSS_5_10, 1)) AND (LAST_RISK_1 < 0 OR LAST_RISK_1 > 20), 0);
LAST_RISK_1_TIME = BARSLAST(RISK_1);
RISK_2 = IF(NOT(IS_UNSUPPORTED), CROSS_5_10 AND LAST_RISK_1_TIME > 0 AND LAST_RISK_1_TIME <= 30, 0);
RISK_2_ONCE = IF(NOT(IS_UNSUPPORTED), RISK_2 AND NOT(REF(RISK_2, 1)), 0);
COUNT_CROSS_2 = IF(NOT(IS_UNSUPPORTED), COUNT_CROSS_5_20 = 2 AND BARSLAST(CROSS_5_20) <= 10, 0);
// 信号绘制
DRAWTEXT(JC1_ONCE AND NOT(IS_UNSUPPORTED), LOW * 0.92, '雷神买点1'), COLORRED;
DRAWICON(JC1_ONCE AND NOT(IS_UNSUPPORTED), LOW * 0.9, 1);
DRAWTEXT(TJ3_POST_ONCE AND NOT(IS_UNSUPPORTED), LOW * 0.92, '雷神买点2'), COLORRED;
DRAWICON(TJ3_POST_ONCE AND NOT(IS_UNSUPPORTED), LOW * 0.9, 1);
DRAWTEXT(TS1_ONCE AND NOT(IS_UNSUPPORTED), HIGH * 1.05, '雷神卖点1'), colorgreen;
DRAWICON(TS1_ONCE AND NOT(IS_UNSUPPORTED), HIGH * 1.07, 2);
DRAWTEXT(TS2_ONCE AND NOT(IS_UNSUPPORTED), HIGH * 1.05, '雷神卖点2'), colorgreen;
DRAWICON(TS2_ONCE AND NOT(IS_UNSUPPORTED), HIGH * 1.07, 2);
DRAWTEXT(RISK_1 AND NOT(IS_UNSUPPORTED), HIGH * 1.05, '风险1'), coloryellow;
DRAWICON(RISK_1 AND NOT(IS_UNSUPPORTED), HIGH * 1.07, 2);
DRAWTEXT(RISK_2_ONCE AND NOT(IS_UNSUPPORTED), HIGH * 1.05, '风险2'), coloryellow;
DRAWICON(RISK_2_ONCE AND NOT(IS_UNSUPPORTED), HIGH * 1.07, 2);
DRAWTEXT(COUNT_CROSS_2 AND NOT(IS_UNSUPPORTED), HIGH * 1.05, '高位风险2'), coloryellow;
DRAWICON(COUNT_CROSS_2 AND NOT(IS_UNSUPPORTED), HIGH * 1.07, 2);
