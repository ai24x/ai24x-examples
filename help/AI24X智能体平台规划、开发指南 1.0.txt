🚨 第七部分：紧急情况处理
7.1 网站无法访问
# 处理步骤:
1. systemctl status ai24x-server  # 检查服务
2. tail -100 /var/log/ai24x/error.log  # 查看日志
3. systemctl restart ai24x-server  # 重启服务
4. ./scripts/rollback.sh previous  # 回滚版本
7.2 数据库问题
# 处理步骤:
1. mongosh --eval "db.adminCommand('ping')"  # 检查连接
2. tail -100 /var/log/mongodb/mongod.log  # 查看日志
3. mongosh --eval "db.repairDatabase()"  # 修复数据库
4. ./scripts/restore-database.sh latest  # 从备份恢复
7.3 安全攻击
# 处理步骤:
1. ufw enable && ufw deny all && ufw allow from your_ip  # 启用防火墙
2. tail -100 /var/log/nginx/access.log | grep -v "200"  # 查看攻击日志
3. ./scripts/block-ip.sh attacker_ip  # 封禁IP
4. ./scripts/notify-users.sh "安全维护中"  # 通知用户
7.4 数据丢失
# 处理步骤:
1. systemctl stop ai24x-server  # 停止写入
2. ./scripts/restore-from-backup.sh  # 从备份恢复
3. ./scripts/verify-data.sh  # 验证数据
4. systemctl start ai24x-server  # 恢复服务
