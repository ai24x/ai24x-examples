import openai
from concurrent.futures import ThreadPoolExecutor

def process_batch(texts, batch_size=10):
    """批量处理文本"""
    results = []
    
    with ThreadPoolExecutor(max_workers=5) as executor:
        futures = []
        for i in range(0, len(texts), batch_size):
            batch = texts[i:i+batch_size]
            future = executor.submit(process_single_batch, batch)
            futures.append(future)
        
        for future in futures:
            results.extend(future.result())
    
    return results

def process_single_batch(batch):
    """处理单个批次"""

        max_tokens=500

2. 流式响应处理

import openai

def stream_response(prompt):
    """流式获取响应"""
    response = openai.ChatCompletion.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],
        stream=True,
        temperature=0.7,
        max_tokens=2000
    )
    
    full_response = ""
    for chunk in response:
        if chunk.choices[0].delta.get("content"):
            content = chunk.choices[0].delta.content
            print(content, end="", flush=True)
            full_response += content
    
    return full_response

# 使用示例
stream_response("写一篇关于人工智能未来的文章")

            model="gpt-4",

            temperature=0.3

        return response.choices[0].message.content

# 代码审查自动化
def code_review(code, language="python"):
    """自动代码审查"""
    prompt = f"""请审查以下{language}代码：
    
    ```{language}
    {code}
    ```
    
    请提供：
    1. 代码质量评分（1-10分）
    2. 潜在问题
    3. 性能优化建议
    4. 安全风险
    5. 最佳实践建议
    6. 重构建议"""
    
    response = openai.ChatCompletion.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],
        temperature=0.2
    )
    
    return response.choices[0].message.content

# 成本监控器
class CostMonitor:
    def __init__(self, budget=100):
        self.budget = budget
        self.used = 0
        self.requests = []
        
    def can_make_request(self, estimated_cost):
        """检查是否超出预算"""
        return self.used + estimated_cost <= self.budget
    
    def record_request(self, tokens_used, cost):
        """记录请求信息"""
        self.used += cost
        self.requests.append({
            "tokens": tokens_used,
            "cost": cost,
            "timestamp": datetime.now()
        })
        
    def get_cost_report(self):
        """生成成本报告"""
        return {
            "total_used": self.used,
            "remaining": self.budget - self.used,
            "request_count": len(self.requests),
            "avg_cost_per_request": self.used / len(self.requests) if self.requests else 0
        }

1. 多轮对话管理

class ConversationManager:
    def __init__(self):
        self.history = []
        self.context_window = 10  # 保留最近10轮对话
        
    def add_message(self, role, content):
        """添加消息到历史"""
        self.history.append({"role": role, "content": content})
        
        # 保持上下文窗口大小
        if len(self.history) > self.context_window * 2:
            self.history = self.history[-self.context_window * 2:]
    
    def get_context(self):
        """获取对话上下文"""
        return self.history.copy()
    
    def summarize_context(self):
        """总结长对话"""
        if len(self.history) < 5:
            return self.history
        
        summary_prompt = "请总结以下对话的主要内容：\n\n"
        for msg in self.history:
            summary_prompt += f"{msg['role']}: {msg['content']}\n"
        
        # 调用ChatGPT生成摘要
        summary = self.call_chatgpt(summary_prompt)
        
        # 用摘要替换旧历史
        self.history = [
            {"role": "system", "content": "对话摘要：" + summary},
            self.history[-1]  # 保留最后一条消息
        ]
        
        return self.history
